"""Place reviewed assets in the slot-4 starter world through the product's own routes.

    python3 place.py --state STATE_JSON ASSET:X_MM:Z_MM[:YAW_MICRORADIANS] ...
    python3 place.py --state STATE_JSON --list

The route pair is the one the browser uses (compositions/preview, then compositions/apply with
the saved-entry binding), copied from lane W's build_world.py. Standard library only. Prints one
line per placement and the world's live objects at the end.
"""

from __future__ import annotations

import argparse
import json
import urllib.error
import urllib.parse
import urllib.request
from pathlib import Path


def call(base: str, token: str, method: str, path: str, body: object | None = None):
    data = None if body is None else json.dumps(body).encode()
    request = urllib.request.Request(
        base + path, data=data, method=method,
        headers={"Authorization": f"Bearer {token}",
                 **({"Content-Type": "application/json"} if data is not None else {})})
    try:
        with urllib.request.urlopen(request, timeout=120) as reply:
            return reply.status, json.loads(reply.read() or b"null")
    except urllib.error.HTTPError as error:
        return error.code, error.read().decode(errors="replace")


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--state", required=True)
    parser.add_argument("--list", action="store_true")
    parser.add_argument("--prefix", default="render-probe")
    parser.add_argument("items", nargs="*")
    args = parser.parse_args()
    state = json.loads(Path(args.state).read_text())
    token = Path(state["token_file"]).read_text().strip()
    base = f"http://127.0.0.1:{state['ports']['api']}"
    status, entry = call(base, token, "POST", "/world-entries/starter", {"title": "My world"})
    if status != 200:
        raise SystemExit(f"starter: {status} {entry}")
    query = "?world_id=" + urllib.parse.quote(entry["world_id"], safe="")
    version_id = entry["authored_version_id"]
    region = entry["authored_scene"]["region"]["region_id"]
    status, version = call(base, token, "GET", f"/world/versions/{version_id}{query}")
    if status != 200:
        raise SystemExit(f"version: {status} {version}")
    base_state, edit_seq, revision = version["state_sha256"], version["edit_seq"], entry["revision"]
    live = [o for o in version["objects"] if not o["removed"]]
    for index, item in enumerate(args.items):
        parts = item.split(":")
        asset, x_mm, z_mm = parts[0], int(parts[1]), int(parts[2])
        yaw = int(parts[3]) if len(parts) > 3 else 3141593
        body = {"source": {"kind": "reviewed_asset", "asset_key": asset},
                "placement": {"subject_id": f"{args.prefix}-{len(live) + index:04d}",
                              "region_id": region,
                              "transform": {"x_mm": x_mm, "y_mm": 0, "z_mm": z_mm,
                                            "yaw_microradians": yaw, "scale_milli": 1000},
                              "origin_role": "fictional", "behaviour": None}}
        status, preview = call(base, token, "POST",
                               f"/world/versions/{version_id}/compositions/preview{query}",
                               {**body, "base_state_sha256": base_state})
        if status != 200 or preview.get("blocked_reason") is not None:
            raise SystemExit(f"preview refused for {item}: {status} {json.dumps(preview)[:600]}")
        status, applied = call(base, token, "POST",
                               f"/world/versions/{version_id}/compositions/apply{query}",
                               {**body, "base_state_sha256": base_state,
                                "saved_entry": {"entry_id": entry["entry_id"],
                                                "base_revision": revision,
                                                "authored_state_sha256": base_state,
                                                "authored_edit_seq": edit_seq}})
        if status not in (200, 201):
            raise SystemExit(f"apply refused for {item}: {status} {json.dumps(applied)[:600]}")
        base_state, edit_seq, revision = applied["state_sha256"], applied["edit_seq"], revision + 1
        print(f"placed {item}", flush=True)
    status, entry = call(base, token, "POST", "/world-entries/starter", {"title": "My world"})
    status, version = call(base, token, "GET", f"/world/versions/{entry['authored_version_id']}{query}")
    for o in version["objects"]:
        if not o["removed"]:
            t = o["transform"]
            print(json.dumps({"object_id": o["object_id"], "asset": o.get("asset", {}).get("asset_key")
                              if isinstance(o.get("asset"), dict) else o.get("asset_key"),
                              "x_mm": t["x_mm"], "z_mm": t["z_mm"], "yaw": t["yaw_microradians"]}))


if __name__ == "__main__":
    main()
