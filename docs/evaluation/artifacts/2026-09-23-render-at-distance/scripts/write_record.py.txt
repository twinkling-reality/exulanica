#: Run files and the record: the evidence gate's own list, and what else identifies this machine
#: or the project's withdrawn name. Scripts name an Authorization header by design, so, as the world scale
#: record's publisher decided, they are held only to the home directory, the withdrawn name and the dash.
FORBIDDEN = (*_FORBIDDEN_TEXT, "ori" + "mera", "file" + "://", chr(0x2014))
FORBIDDEN_IN_SCRIPTS = (str(Path.home()), "ori" + "mera", chr(0x2014))


def sanitise(text: str, name: str = "", script: bool = False) -> str:
    for old, new in SUBSTITUTIONS:
        text = text.replace(old, new)
    text = re.sub(r"/var/folders/[^/\s\"']+/[^/\s\"']+/T", "<TMPDIR>", text)
    refused = [phrase for phrase in (FORBIDDEN_IN_SCRIPTS if script else FORBIDDEN) if phrase in text]
    if refused:
        where = text.index(refused[0])
        raise SystemExit(f"{name} still carries {refused[0]!r}: {text[max(0, where - 60):where + 60]!r}")
    return text


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def load_slotted(path: Path) -> dict:
    """A runner's JSON with the slot wrapper's own lines (gpu-slot: ...) removed."""
    lines = [line for line in path.read_text().splitlines() if not line.startswith(("gpu-slot:", "quiet-slot:"))]
    return json.loads("\n".join(lines))


def millionths(value: float) -> int:
    return round(value * 1_000_000)


def mm(value: float) -> int:
    return round(value * 1000)


published: list[dict] = []


def publish(source: Path, target: str, *, text: bool = True, script: bool = False) -> dict:
    """Copy one file beside the record and bind it by digest."""
    destination = ARTIFACTS / target
    destination.parent.mkdir(parents=True, exist_ok=True)
    if text:
        destination.write_text(sanitise(source.read_text(), source.name, script))
    else:
        shutil.copyfile(source, destination)
    entry = {"path": f"{REL_ARTIFACTS}/{target}", "byte_size": destination.stat().st_size,
             "sha256": sha(destination)}
    published.append(entry)
    return entry


def publish_json(document: object, target: str) -> dict:
    destination = ARTIFACTS / target
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.write_text(sanitise(json.dumps(document, indent=1) + "\n", target))
    entry = {"path": f"{REL_ARTIFACTS}/{target}", "byte_size": destination.stat().st_size,
             "sha256": sha(destination)}
    published.append(entry)
    return entry


# -- the band cause (dev server, tree e9dee3c2 unchanged, before any product change) ----------------

def band_cause() -> dict:
    arms_run = json.loads(sorted((RUNS / "probe-bands").glob("*.json"))[0].read_text())
    publish(sorted((RUNS / "probe-bands").glob("*.json"))[0], "runs/band-cause/arms.json")
    arms = {}
    for arm, rows in arms_run["arms"].items():
        arms[arm] = [{"mark_m": row["mark"], "sky_distinct_colours": row["sky_distinct"],
                      "lit_face_differing_pixels": row["face_differing"], "lit_face_middle_row_steps": row["face_steps"],
                      "camera_render_z_mm": mm(row["camera"]["render"][2])} for row in rows]
    internals = sorted((RUNS / "probe-internals").glob("*.json"))
    shadow_camera, casters = {}, {}
    for path in internals:
        run = json.loads(path.read_text())
        lists_casters = "casters" in run["rows"][0]["dump"]
        publish(path, f"runs/band-cause/{'casters' if lists_casters else 'shadow-camera'}.json")
        for row in run["rows"]:
            dump = row["dump"]
            if lists_casters:
                casters[row["label"]] = {
                    "shadow_depth_range_mm": mm(dump["shadowCamera"]["far"]),
                    "casters": [{"node": c["node"], "min_mm": [mm(v) for v in c["min"]], "max_mm": [mm(v) for v in c["max"]]}
                                for c in dump["casters"]]}
            else:
                shadow_camera[row["label"]] = {
                    "shadow_depth_range_mm": mm(dump["shadowCamera"]["far"]),
                    "camera_render_z_mm": mm(dump["camera"][14]),
                    "render_root_z_mm": mm(dump["renderRoot"][2])}
    shadow_path = sorted((RUNS / "probe-shadow").glob("*.json"))[0]
    publish(shadow_path, "runs/band-cause/shadow-arms.json")
    shadow = json.loads(shadow_path.read_text())
    shadow_arms = {arm: {"applied": data["report"], "marks": [
        {"mark_m": row["mark"], "self_shadow_pixels": row["self"], "ground_shadow_pixels": row["ground"],
         "shadow_touching_object_pixels": row["touching"], "ground_shadow_top_row": row["ground_top_row"],
         "lit_face_middle_row_steps": row["face_steps"], "sky_distinct_colours": row["sky_distinct"]}
        for row in data["rows"]]} for arm, data in shadow["arms"].items()}
    for arm in shadow_arms.values():
        applied = arm["applied"]
        arm["applied"] = {key: (millionths(value) if isinstance(value, float) else value)
                          for key, value in applied.items() if key != "before"}
        arm["applied"]["before_milli"] = {k: round(v * 1000) for k, v in applied["before"].items()}
    return {"arms": arms, "shadow_camera": shadow_camera, "casters": casters, "shadow_arms": shadow_arms}


# -- the plate on the ground ---------------------------------------------------------------------------

def plate() -> dict:
    out = {}
    for path in sorted((RUNS / "probe-plate").glob("record-*-0c1abce8d3c4.json")) + \
            sorted((RUNS / "probe-plate").glob("arm-*.json")):
        run = json.loads(path.read_text())
        publish(path, f"runs/plate/{path.name}")
        grounds = {}
        for key, value in run["grounds"].items():
            grounds[key] = {"poses": value["poses"], "poses_with_an_interior": value["poses"] - value["footprintless"],
                            "drawn_min_millionths": millionths(value["min"]),
                            "drawn_mean_millionths": millionths(value["mean"]),
                            "poses_drawn_under_99_percent": value["under_0_99"],
                            "clean_min_millionths": millionths(value["clean_min"]),
                            "clean_mean_millionths": millionths(value["clean_mean"]),
                            "poses_clean_under_99_percent": value["clean_under_0_99"]}
        out[path.name[:-len(".json")]] = {"arm": {"legacy": run.get("legacy", False), "bias": run.get("bias"),
                                                  "face_half_m": run.get("faceHalf"), "face_grid": run.get("faceGrid")},
                                          "grounds": grounds}
    return out


# -- the GPU pixel scenes -----------------------------------------------------------------------------

PIXEL_RUNS = [
    ("baseline-e9dee3c2", "the tree at e9dee3c2 before any change, with objects placed after the first frame"),
    ("sky-bias", "after the sky and the starter's shadow bias, and the object residency fix"),
    ("fixes-1", "after the plate's depth order and the endless face grid as well: the reference for 5a and 5d"),
    ("step-5a", "after the world-kind module (5a), meant to move no pixel"),
    ("step-5d", "after the atmosphere module (5d), meant to move no pixel"),
    ("final", "the delivered tree, which differs from step-5d by comments only"),
]


def pixels() -> dict:
    runs = {}
    for name, what in PIXEL_RUNS:
        document = load_slotted(RUNS / "pixels" / f"{name}.json")
        publish_json(document, f"runs/pixels/{name}.json")
        runs[name] = {"what": what, "browser": document["browser"], "renderer": document["renderer"],
                      "repeats": document["repeats"], "identical_across_repeats": document["identical_across_repeats"],
                      "scenes": {s["scene"]: {"sha256": s["sha256"], "drawn_pixels": s["drawnPixels"],
                                              **({"sky_colours": s["skyColours"]} if "skyColours" in s else {})}
                                 for s in document["scenes"]},
                      "plate_sweeps": [{"ground": p["ground"], "poses": p["poses"], "measured": p["measured"],
                                        "drawn_min_millionths": millionths(p["drawnMin"]),
                                        "clean_min_millionths": millionths(p["cleanMin"]),
                                        "under_whole": p["underWhole"]}
                                       for p in (document.get("plate_sweeps") or [])]}
    reference = runs["fixes-1"]["scenes"]
    for step in ("step-5a", "step-5d"):
        runs[step]["byte_identical_to_fixes_1"] = sum(
            1 for scene, value in runs[step]["scenes"].items() if value["sha256"] == reference[scene]["sha256"])
    runs["final"]["byte_identical_to_step_5d"] = sum(
        1 for scene, value in runs["final"]["scenes"].items() if value["sha256"] == runs["step-5d"]["scenes"][scene]["sha256"])
    baseline = runs["baseline-e9dee3c2"]["scenes"]
    final = runs["final"]["scenes"]
    runs["moved_from_baseline"] = sorted(s for s in final if final[s]["sha256"] != baseline[s]["sha256"])
    runs["unchanged_from_baseline"] = sorted(s for s in final if final[s]["sha256"] == baseline[s]["sha256"])
    return runs


# -- the tile load timeouts ------------------------------------------------------------------------------

def tile_flake() -> dict:
    out = {}
    for n, what in ((1, "before: whole web run, default 5 s timeout"), (2, "after: whole web run, VERIFIED_LOAD_TIMEOUT_MS")):
        log = RUNS / "web" / f"web-whole-{n}.log.txt"
        text = log.read_text()
        tests = {}
        for line in text.splitlines():
            match = re.match(r"^ +([✓×]) packages/atlas-react/test/generated-tile-runtime\.test\.ts > a neighbouring tile reaching the runtime > (.+?) (\d+)ms$", line)
            if match:
                tests[match.group(2)] = {"passed": match.group(1) == "✓", "milliseconds": int(match.group(3))}
        totals = [line.strip() for line in text.splitlines() if line.strip().startswith(("Test Files", "Tests "))]
        out[f"run_{n}"] = {"what": what, "log_sha256": sha(log), "log_bytes": log.stat().st_size,
                           "neighbour_tests": tests, "totals": totals}
    return out


# -- the release-build distance series ------------------------------------------------------------------

def distance_series() -> dict:
    series_path = DIST / "runs" / "distance" / "series.json"
    if not series_path.exists():
        return {"status": "not run"}
    series = json.loads(series_path.read_text())
    folder = DIST / "runs" / "distance"
    for path in sorted(folder.glob("distance-*.json")):
        publish(path, f"runs/distance/{path.name}")
    # Sessions the idle gate threw out: bound above like every other run file, counted by nothing.
    series["sessions_not_accepted"] = [path.name for path in sorted(folder.glob("distance-*.json"))
                                       if path.name.endswith(("-discarded.json", "-invalid.json", "-stopped.json"))]
    publish(folder / "shading.json", "runs/distance/shading.json")
    publish(series_path, "runs/distance/series.json")
    # The captures of sessions the idle gate discarded, read by shading_not_timed.py: pixels only.
    not_timed = folder / "shading-not-timed.json"
    if not_timed.exists():
        publish(not_timed, "runs/distance/shading-not-timed.json")
        series["pixels_not_timed"] = [row for row in json.loads(not_timed.read_text()) if row["mark_m"] != 0]
    gates = DIST / "runs" / "gates.jsonl"
    if gates.exists():
        publish(gates, "runs/distance/gates.jsonl.txt")
    return series

def machine() -> dict:

    return {"hardware_model": run("sysctl", "-n", "hw.model"), "logical_cpus": int(run("sysctl", "-n", "hw.logicalcpu")),
            "memory_bytes": int(run("sysctl", "-n", "hw.memsize")), "os": f"macOS {run('sw_vers', '-productVersion')}",
            "node": run("node", "--version"), "python": platform.python_version()}



# -- the binding harness ---------------------------------------------------------------------------------

def harness() -> dict:
    suites = json.loads((RUNS / "web" / "lane-suites.json").read_text())
    files = {}
    for result in suites["testResults"]:
        name = "web/" + result["name"].split("/web/", 1)[1]
        files[name] = {"tests": len(result["assertionResults"]),
                       "passed": sum(1 for a in result["assertionResults"] if a["status"] == "passed")}
    breaks = {}
    for log in sorted((RUNS / "break").glob("*.log.txt")) if (RUNS / "break").exists() else []:
        entry = publish(log, f"runs/break/{log.name}")
        lines = [line for line in log.read_text().splitlines() if ": exit " in line]
        breaks[log.name] = {"mutants": len(lines), "caught": sum(1 for line in lines if ": exit 1 " in line or line.endswith(": exit 1")),
                            "sha256": entry["sha256"]}
    ansi = re.compile(r"\x1b\[[0-9;]*m")

    def totals(name: str) -> dict:
        log = RUNS / "web" / name
        text = ansi.sub("", log.read_text())
        return {"log_sha256": sha(log), "log_bytes": log.stat().st_size,
                "totals": [line.strip() for line in text.splitlines() if line.strip().startswith(("Test Files", "Tests "))]}

    package = publish(RUNS / "web" / "atlas-react-package.log.txt", "runs/web/atlas-react-package.log.txt")
    return {"focused_suites_on_the_delivered_tree": {"suites": files, "tests": suites["numTotalTests"],
                                                     "passed": suites["numPassedTests"], "failed": suites["numFailedTests"],
                                                     "skipped": suites["numPendingTests"]},
            "atlas_react_package_on_the_delivered_tree": {**totals("atlas-react-package.log.txt"), "published": package["path"]},
            "typecheck_and_boundaries_on_the_delivered_tree": {
                "typecheck_log_sha256": sha(RUNS / "web" / "typecheck.log.txt"),
                "typecheck": [line for line in (RUNS / "web" / "typecheck.log.txt").read_text().splitlines() if " exit " in line],
                "boundaries": [line.strip() for line in (RUNS / "web" / "boundaries.log.txt").read_text().splitlines()
                               if "dependency violations" in line]},
            "break_checks": breaks,
            "whole_web_run": {**totals("web-whole-3.log.txt"),
                              "tree": "the product tree before a comments-only edit to atmosphere.ts and world-field.ts; the release bundle built from both trees has the same content hash"}}


# -- the words: every number in them is read from the record built above ------------------------------

def span(values) -> str:
    values = [v for v in values if v is not None]
    low, high = min(values), max(values)
    return f"{low:,}" if low == high else f"{low:,} to {high:,}"


def distance_marks(record: dict) -> dict:
    return {row["mark_m"]: row for row in record["distance"]["series"].get("marks", [])}


def captures(record: dict) -> list[dict]:
    folder = DIST / "runs" / "distance"
    sessions = sorted(row["file"] for row in record["distance"]["series"].get("pixels_not_timed", [])
                      if row["mark_m"] == 8000)
    if not sessions:
        return []
    stem = sessions[0][:-len("-discarded.json")]
    bound = []
    for mark, name, what in (
            (0, "arrival-with-reference.png", "The arrival point of a release-build 8 km session on the changed tree, with its reference cube before the camera: the sky above the horizon is a gradient and the lit face is smooth. The idle gate discarded this session for timing; its pixels are engine renders the load does not change."),
            (8000, "8km-with-reference.png", "The same session 8 km along the sprint with a new reference cube at the same relative pose: the sky is the same gradient and the lit face carries no band.")):
        source = folder / f"{stem}-mark-{mark}-with.png"
        entry = publish(source, f"captures/{name}", text=False)
        bound.append({**entry, "as_run": source.name, "what": what})
    for scene, what in (("flat-plate", "A reviewed marker plate on a bounded starter ground, GPU pixel scene, before (left: the metre marks drawn over it) and after (right: the plate over them)."),
                        ("endless-cube-after-map", "A cube placed on an endless starter ground and then the Map opened and closed, GPU pixel scene, before (left: the cube is gone) and after (right: the cube is drawn).")):
        from PIL import Image
        before = Image.open(RUNS / "pixels" / "baseline-e9dee3c2" / f"{scene}.png").convert("RGB")
        after = Image.open(RUNS / "pixels" / "fixes-1" / f"{scene}.png").convert("RGB")
        pair = Image.new("RGB", (before.width + after.width + 8, before.height), (255, 255, 255))
        pair.paste(before, (0, 0))
        pair.paste(after, (before.width + 8, 0))
        target = LANE / "runs" / "pixels" / f"{scene}-before-after.png"
        pair.save(target, optimize=False)
        entry = publish(target, f"captures/{scene}-before-after.png", text=False)
        bound.append({**entry, "as_run": f"baseline-e9dee3c2/{scene}.png beside fixes-1/{scene}.png", "what": what})
    return bound
