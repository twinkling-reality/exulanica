#!/usr/bin/env python3
"""Does swapping the open model that decides for twelve of a town's people change how they fare?

    measure_town_comparison_v2.py preregister --name <record name> --slot-state <file> \\
        --world <world id> --version <uuid> --control <uuid> [...] --bound-usd <amount> \\
        [--predecessor <comparison uuid>] [--predecessor-record <path>]
    measure_town_comparison_v2.py run --name <record name> --slot-state <file> --seeds <file> \\
        --dotenv <file>
    measure_town_comparison_v2.py record --name <record name> --slot-state <file> \\
        --comparison <uuid> --notes <file>

The third version of this script writes every record under the outer profile the retained records
carry (``exulanica.digest-bound-record/v1``), what the record is as its own ``kind``, and can carry
a record written under another outer profile under it, its record unchanged (``rewrap``). The
second version of this script. The first raised while writing its record (a float, the run's
seconds, which a record refuses) and let the process's default call ceiling stop the run; this
one writes every figure as an integer or a Decimal string, writes a record from a comparison's
stored runs (``record``), which the run step writes through the same code, states the process
call ceiling in the pre-registration and sets it, and admits a seed only while both the bound and
the call ceiling hold it.

**The question.** A town made from the ``small_town`` recipe, its people brought into its
purposeful society, one simulated hour. The first twelve of its people in the order the plan route
lists them (by name, then identity) are the group; everybody else keeps their routine in every arm.
With the group decided by Nemotron 3 Nano 30B, run twice so its second run bounds run-to-run
variation, or by Nemotron 3.5 Lightning, how did the group's people fare under the third score of
a person in a world (``society-person-score.v3``: half the need they were spared, half how many
different kinds of thing they did, each as a share of what their own routine gives them against
waiting on the same seed), and is Lightning's difference from Nano larger than the protocol lets
a comparison claim and than the control pair's?

**The pre-registration** (``preregister``) states the comparison before any held-out seed is run:
the town (its recipe, its tiles' inputs by digest, the society input the comparison freezes by
sequence and digest, its people), the group and how it was chosen, the candidates and how each is
asked, the arms and the claim, the held-out seeds by digest (``society-comparison-seeds.v4``), the
window, the protocol's values, the verdict rule, the least bound that lets every seed finish under
the host's admission between seeds (what the plan route derives for it) and the bound, the positive
control's development comparison read from its stored runs with every run's integer terms, the
scoring binding and the tree it measures, every path that differs from HEAD by its digest.

**The run** (``run``) is judged once: it refuses when its record exists, when the tree is not the
registered one apart from ``docs/evaluation/``, and when the bound in ``BUDGET_VARIABLE`` is not the
registered one. It reads the held-out seeds from ``--seeds``, uses one only when the SHA-256 of its
text is the catalog's commitment, and never prints or records one. It plays the anchors first, then
each seed's model runs, admitting a seed only while what is left of the bound holds what one like
it typically costs and what its runs can hold reserved at once, as a host does; draws every
completed run; and writes the record from the stored runs: every run's integer terms, each term,
each score, the verdict, the spend by model and the time a stored drawing takes to read.

**Spend.** The key is read from the operator's ``.env`` inside this process only, by
:func:`exulanica.models.credentials.api_key_from_env`, and reaches nothing but the client. Every
call of every run is within the bound.
"""

from __future__ import annotations

import argparse
import datetime as dt
import hashlib
import json
import os
import secrets
import subprocess
import sys
import time
import uuid
from collections.abc import Mapping, Sequence
from decimal import ROUND_CEILING, Decimal
from pathlib import Path
from typing import Any, Final

ROOT: Final = Path(__file__).resolve().parents[2]
BUDGET_VARIABLE: Final = "EXULANICA_BUDGET_USD"
CALLS_VARIABLE: Final = "EXULANICA_BUDGET_MAX_CALLS"
#: How much above the plan's most calls the pre-registered process call ceiling is set: the most is
#: already every ask answered as often as the contract allows, and a tenth more leaves room for
#: whatever else the process's client is asked.
CALL_MARGIN: Final = Decimal("1.1")
#: How long after registration a held-out run may start, in hours.
RUN_WINDOW_HOURS: Final = 12


def _paths(name: str) -> dict[str, str]:
    """The records and as-run copies a comparison named ``name`` writes."""
    return {
        "preregistration": f"docs/evaluation/{name}-preregistration.json",
        "record": f"docs/evaluation/{name}.json",
        "script_as_run": f"docs/evaluation/artifacts/{name}/measure_town_comparison-as-run.py.txt",
    }


#: The candidates, in the order the claim reads them: the first is run again as the control.
MODELS: Final = (
    ("nebius_token_factory", "nvidia/NVIDIA-Nemotron-3-Nano-30B-A3B"),
    ("nebius_token_factory", "nvidia/Nemotron-3_5-Lightning"),
)
#: The group: the first this many of the town's people in the plan route's order.
GROUP_SIZE: Final = 12
RECIPE: Final = "small_town"


def _sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def _now() -> str:
    return dt.datetime.now(dt.UTC).replace(microsecond=0).isoformat()


#: The outer profile every retained record under docs/evaluation carries
#: (tests/test_retained_evaluation_records.py); what a record is, is its own "kind".
OUTER_PROFILE: Final = "exulanica.digest-bound-record/v1"


def _sealed(record: dict[str, Any], kind: str) -> dict[str, Any]:
    from exulanica.canonical import canonical_json

    inner = {"kind": kind, **record}
    return {
        "profile": OUTER_PROFILE,
        "record": inner,
        "record_sha256": _sha256(canonical_json(inner)),
    }


def rewrap(arguments: argparse.Namespace) -> None:
    """Carry a record written under another outer profile under :data:`OUTER_PROFILE`, its record
    and its record_sha256 unchanged, and print both files' digests."""
    from exulanica.canonical import canonical_json

    path = ROOT / arguments.path
    before = path.read_bytes()
    document = json.loads(before)
    if _sha256(canonical_json(document["record"])) != document["record_sha256"]:
        raise SystemExit("the record does not reproduce its digest")
    rewrapped = {**document, "profile": OUTER_PROFILE}
    after = (json.dumps(rewrapped, indent=2, sort_keys=True) + "\n").encode("utf-8")
    path.write_bytes(after)
    print(
        json.dumps(
            {
                "path": arguments.path,
                "outer_profile_was": document["profile"],
                "file_sha256_before": _sha256(before),
                "file_sha256_after": _sha256(after),
                "record_sha256": document["record_sha256"],
            }
        )
    )


def _write_new(relative: str, value: dict[str, Any]) -> None:
    path = ROOT / relative
    if path.exists():
        raise SystemExit(f"{relative} exists: a record is written once")
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n", encoding="utf-8")


def _tree() -> dict[str, Any]:
    """HEAD and the digest of every path that differs from it, tracked or not, but the records."""
    head = subprocess.run(
        ["git", "rev-parse", "HEAD"], cwd=ROOT, capture_output=True, text=True, check=True
    ).stdout.strip()
    listed = subprocess.run(
        ["git", "status", "--porcelain", "--untracked-files=all"],
        cwd=ROOT,
        capture_output=True,
        text=True,
        check=True,
    ).stdout.splitlines()
    files = {}
    for line in listed:
        relative = line[3:].strip()
        if relative.startswith("docs/evaluation/") or relative.startswith(".exulanica/"):
            continue
        path = ROOT / relative
        files[relative] = _sha256(path.read_bytes()) if path.is_file() else None
    return {"head": head, "files_sha256": dict(sorted(files.items()))}


def _environment(
    state: Mapping[str, Any], budget_usd: str, dotenv: Path, max_calls: int | None = None
) -> None:
    """This process's configuration as the slot's API has it, the model key read from the
    operator's ``.env`` into this process alone and never printed."""
    from exulanica.models.credentials import api_key_from_env

    urls = json.loads((Path(state["run_dir"]) / "api-urls.json").read_text(encoding="utf-8"))
    os.environ.update({key: str(value) for key, value in urls.items()})
    os.environ["EXULANICA_DATA_DIR"] = state["data_dir"]
    os.environ["EXULANICA_SOCIETY_CONTROL_WORKSPACES"] = json.dumps([state["workspace_id"]])
    os.environ["EXULANICA_EGRESS_ALLOWLIST"] = json.dumps(["https://api.tokenfactory.nebius.com"])
    os.environ[BUDGET_VARIABLE] = budget_usd
    if max_calls is not None:
        os.environ[CALLS_VARIABLE] = str(max_calls)
    # The application's services need a token directory; this process serves no request, so it
    # holds one random token of its own, granted to read the world alone.
    os.environ["EXULANICA_API_TOKENS"] = json.dumps(
        {
            secrets.token_urlsafe(48): {
                "workspace_id": state["workspace_id"],
                "actor": state["actor"],
                "permissions": ["world.read"],
            }
        }
    )
    os.environ["NEBIUS_API_KEY"] = api_key_from_env("NEBIUS_API_KEY", dotenv=dotenv)


def _owner(state: Mapping[str, Any]) -> Any:
    import psycopg
    from psycopg.rows import dict_row

    return psycopg.connect(state["database"]["owner_url_for_evidence_reads"], row_factory=dict_row)


def _society(state: Mapping[str, Any], version: uuid.UUID, world_id: str) -> dict[str, Any]:
    """The town's society as the database holds it: its people, in the plan route's order."""
    with _owner(state) as connection:
        society = connection.execute(
            "select society_id, state from world_society where workspace_id=%s and world_id=%s "
            "and version_id=%s",
            (state["workspace_id"], world_id, version),
        ).fetchone()
    people = sorted(
        ({"id": p["id"], "name": p["display_name"]} for p in society["state"]["inhabitants"]),
        key=lambda person: (person["name"], person["id"]),
    )
    return {"society_id": str(society["society_id"]), "people": people}


def _frozen_input(state: Mapping[str, Any], world_id: str, comparison_id: str) -> dict[str, Any]:
    """The society input a comparison froze, by sequence and digest, as its definition holds it."""
    with _owner(state) as connection:
        row = connection.execute(
            "select input_seq, input_sha256 from society_comparison where world_id=%s "
            "and comparison_id=%s",
            (world_id, comparison_id),
        ).fetchone()
    return {"input_seq": row["input_seq"], "document_sha256": row["input_sha256"]}


def _stored_runs(state: Mapping[str, Any], world_id: str, comparison_id: str) -> dict[str, Any]:
    """A comparison's runs from what they stored, each scored under its definition's score from
    its integer terms (the verdict module's own reading), with the terms beside every score."""
    from exulanica.world.society_comparison_result import comparison_result

    with _owner(state) as connection:
        row = connection.execute(
            "select comparison_id, created_at, document from society_comparison "
            "where world_id=%s and comparison_id=%s",
            (world_id, comparison_id),
        ).fetchone()
        runs = connection.execute(
            "select r.run_id, r.arm, r.seed_digest, o.status, o.document as outcome "
            "from society_comparison_run r left join society_comparison_outcome o "
            "using (workspace_id, world_id, comparison_id, run_id) "
            "where r.world_id=%s and r.comparison_id=%s",
            (world_id, comparison_id),
        ).fetchall()
    return comparison_result(row, runs, model_name=lambda model_id: model_id)


def _spend(state: Mapping[str, Any], world_id: str, comparison_id: str) -> list[dict[str, Any]]:
    """Every paid attempt of the comparison by provider and model, from its receipts."""
    with _owner(state) as connection:
        rows = connection.execute(
            "select d.receipt->'provider' as provider from society_comparison_decision d "
            "where d.world_id=%s and d.comparison_id=%s",
            (world_id, comparison_id),
        ).fetchall()
    totals: dict[tuple[str, str], dict[str, Any]] = {}
    for row in rows:
        call = row["provider"] or {}
        key = (str(call.get("provider")), str(call.get("model_id")))
        held = totals.setdefault(key, {"asks": 0, "cost_usd": Decimal(0), "cost_unknown": 0})
        held["asks"] += 1
        if call.get("cost_usd") is None:
            held["cost_unknown"] += 1
        else:
            held["cost_usd"] += Decimal(str(call["cost_usd"]))
    return [
        {
            "provider": p,
            "model_id": m,
            "asks": v["asks"],
            "cost_usd": str(v["cost_usd"]),
            "cost_unknown": v["cost_unknown"],
        }
        for (p, m), v in sorted(totals.items())
    ]


def _plan_figures(
    api: Any, version: uuid.UUID, world_id: str, people: Sequence[str], seeds: int
) -> dict[str, Any]:
    query = [
        ("world_id", world_id),
        ("role", "society_decision"),
        ("group", "named"),
        ("control", "true"),
        ("seeds", str(seeds)),
        *(("model", f"{provider}/{model_id}") for provider, model_id in MODELS),
        *(("person", person) for person in people),
    ]
    return api(f"/world/versions/{version}/society/comparisons/plan", query)["plan"]


def _api(state: Mapping[str, Any]) -> Any:
    import urllib.parse
    import urllib.request

    token = Path(state["token_file"]).read_text(encoding="utf-8").strip()

    def get(path: str, query: Sequence[tuple[str, str]]) -> dict[str, Any]:
        request = urllib.request.Request(
            f"http://127.0.0.1:{state['ports']['api']}{path}?{urllib.parse.urlencode(query)}",
            headers={"Authorization": f"Bearer {token}"},
        )
        with urllib.request.urlopen(request, timeout=300) as response:
            return json.loads(response.read())

    return get


def preregister(arguments: argparse.Namespace) -> None:
    from exulanica.world.society_catalogs import load_comparison_catalogs
    from exulanica.world.society_comparison_result import scoring_binding
    from exulanica.world.society_comparison_verdict import protocol_values

    state = json.loads(arguments.slot_state.read_text(encoding="utf-8"))
    world_id = arguments.world
    api = _api(state)
    town = _society(state, arguments.version, world_id)
    group = town["people"][:GROUP_SIZE]
    entries = api("/world-entries", [])
    ground = next(entry for entry in entries if entry["world_id"] == world_id)["generated_ground"]
    catalogs = load_comparison_catalogs()
    held_out = [
        {"key": key, "seed_digest": str(entry["seed_digest"])}
        for key, entry in catalogs.seeds.items()
        if entry["phase"] == "held_out"
    ]
    ids = [person["id"] for person in group]
    one, every = (
        _plan_figures(api, arguments.version, world_id, ids, 1),
        _plan_figures(api, arguments.version, world_id, ids, len(held_out)),
    )
    least = Decimal(every["typical_usd"]) + Decimal(one["held_usd"])
    call_ceiling = int(
        (Decimal(every["calls_most"]) * CALL_MARGIN).to_integral_value(ROUND_CEILING)
    )
    bound = Decimal(arguments.bound_usd)
    if bound < least:
        raise SystemExit(f"the bound {bound} is under the least that lets every seed finish")
    controls = []
    pooled: dict[str, list[Decimal]] = {}
    for comparison_id in arguments.control:
        found = _stored_runs(state, world_id, comparison_id)
        (seed,) = found["seeds"]
        scores = {arm: Decimal(run["score"]) for arm, run in seed["runs"].items() if run["score"]}
        for arm in ("model_a", "model_b", "model_a_again"):
            pooled.setdefault(arm, []).append(scores[arm])
        controls.append(
            {
                "comparison_id": comparison_id,
                "seed": seed["name"],
                "phase": found["phase"],
                "score_version": found["score_version"],
                "runs": {
                    arm: {
                        "status": run["status"],
                        "score": run["score"],
                        "parts": run["parts"],
                        "terms": run["terms"],
                    }
                    for arm, run in seed["runs"].items()
                },
                "primary_difference": str(abs(scores["model_b"] - scores["model_a"])),
                "control_difference": str(abs(scores["model_a_again"] - scores["model_a"])),
                "spend": _spend(state, world_id, comparison_id),
            }
        )
    means = {arm: sum(values) / len(values) for arm, values in pooled.items()}
    record = {
        "question": __doc__.split("\n\n")[2],
        "written_before_any_held_out_call": True,
        "town": {
            "recipe": RECIPE,
            "world_id": world_id,
            "version_id": str(arguments.version),
            "ground": ground,
            "society_id": town["society_id"],
            "frozen_input": _frozen_input(state, world_id, arguments.control[0]),
            "population": len(town["people"]),
        },
        "group": {
            "rule": f"the first {GROUP_SIZE} of the town's people in the plan route's "
            "order, by name then identity; everybody else follows their routine",
            "people": group,
        },
        "models": [{"provider": p, "model_id": m} for p, m in MODELS],
        "arms": {
            "model_a": "Nemotron 3 Nano 30B",
            "model_b": "Nemotron 3.5 Lightning",
            "model_a_again": "Nemotron 3 Nano 30B, the control",
            "routine": "one",
            "wait": "zero",
        },
        "claim": {
            "primary": ["model_a", "model_b"],
            "family": [["model_a", "model_b"], ["routine", "model_a"], ["routine", "model_b"]],
            "control": ["model_a", "model_a_again"],
        },
        "held_out_seeds": held_out,
        "catalogs": dict(catalogs.versions),
        "protocol": protocol_values(catalogs),
        "verdict_rule": (
            "Pooled over seeds, as the server's verdict "
            "(exulanica/world/society_comparison_verdict.py and "
            "exulanica/world/society_comparison_claim.py under binding v3). For each registered "
            "pair, the per-seed differences of the v3 score over the seeds both arms scored, "
            "their mean and a percentile bootstrap interval (bootstrap_resamples, "
            "interval_per_mille); Holm's procedure over the family at family_alpha_per_mille. "
            "The control's bound is the larger magnitude of the two ends of the control pair's "
            "(Nano, Nano again) interval. The verdict is different, naming the higher arm, only "
            "where Holm rejects the primary difference (Nano, Lightning) and the magnitude of its "
            "mean exceeds the control's bound; otherwise no_measured_difference. Per-seed "
            "differences are reported beside it and never decide it. A comparison with any run "
            "missing or failed, including a seed the bound did not admit "
            "(comparison_bound_before_seed), reads incomplete and claims nothing; one with fewer "
            "than two seeds every judged arm scored reads not_judged (too_few_seeds_scored); a "
            "seed excluded by name (need_below_floor, variety_not_spared) carries no score."
        ),
        "run_window": {
            "simulated_minutes": protocol_values(catalogs)["window_ticks"],
            "starts_before_utc": (
                dt.datetime.now(dt.UTC).replace(microsecond=0)
                + dt.timedelta(hours=RUN_WINDOW_HOURS)
            ).isoformat(),
        },
        "process_calls": {
            "calls_most": every["calls_most"],
            "process_call_ceiling": call_ceiling,
            "rule": "the run's process may make at most this many calls "
            f"({CALLS_VARIABLE}), the plan's most calls for every run and a tenth more; a seed "
            "is admitted only while both what is left of the bound and what is left of the "
            "call ceiling hold what its runs can take",
        },
        "predecessor": None
        if arguments.predecessor is None
        else {
            "comparison_id": arguments.predecessor,
            "record": arguments.predecessor_record,
            "outcome": _served(state, world_id, arguments.version, arguments.predecessor)[
                "verdict"
            ],
        },
        "bound": {
            "bound_usd": str(bound),
            "least_every_seed_finishes_usd": str(least),
            "typical_usd": every["typical_usd"],
            "held_usd": one["held_usd"],
            "typical_record": every["typical_record"],
        },
        "positive_control": {
            "rule": "the judged score must separate the development town runs of this design "
            "beyond the control pair: pooled over the development seeds, the Lightning and Nano "
            "difference exceeds the Nano pair's",
            "comparisons": controls,
            "pooled_means": {arm: str(value) for arm, value in means.items()},
            "pooled_primary_difference": str(abs(means["model_b"] - means["model_a"])),
            "pooled_control_difference": str(abs(means["model_a_again"] - means["model_a"])),
            "note": "scores as served, four places; every run's integer terms are beside them",
        },
        "script_sha256": _sha256(Path(__file__).read_bytes()),
        "script_as_run": _paths(arguments.name)["script_as_run"],
        "scoring": scoring_binding(catalogs),
        "tree": _tree(),
        "registered_utc": _now(),
    }
    written = _paths(arguments.name)["preregistration"]
    _write_new(written, _sealed(record, "exulanica.town-comparison-preregistration/v1"))
    print(
        json.dumps(
            {
                "written": written,
                "least_usd": str(least),
                "bound_usd": str(bound),
                "process_call_ceiling": call_ceiling,
            }
        )
    )


def _served(state: Mapping[str, Any], world_id: str, version: Any, comparison_id: str) -> Any:
    """A comparison's result as the application's route serves it."""
    return _api(state)(
        f"/world/versions/{version}/society/comparisons/{comparison_id}", [("world_id", world_id)]
    )


def _database_facts(state: Mapping[str, Any], world_id: str, comparison_id: str) -> dict[str, Any]:
    """What the database recorded of a comparison's playing: when it was defined, when its first
    and last outcome were recorded, how many asks and attempts its receipts hold, its runs by
    status and failure code, and how many of its runs are drawn."""
    with _owner(state) as connection:
        defined = connection.execute(
            "select created_at from society_comparison where world_id=%s and comparison_id=%s",
            (world_id, comparison_id),
        ).fetchone()["created_at"]
        span = connection.execute(
            "select min(recorded_at) as first, max(recorded_at) as last "
            "from society_comparison_outcome where world_id=%s and comparison_id=%s",
            (world_id, comparison_id),
        ).fetchone()
        asks = connection.execute(
            "select count(*) as asks, coalesce(sum(coalesce(jsonb_array_length("
            "receipt->'provider'->'attempts'), 1)), 0) as attempts "
            "from society_comparison_decision where world_id=%s and comparison_id=%s",
            (world_id, comparison_id),
        ).fetchone()
        statuses = connection.execute(
            "select r.arm, coalesce(o.status, 'open') as status, o.document->>'code' as code, "
            "count(*) as runs from society_comparison_run r left join society_comparison_outcome o "
            "using (workspace_id, world_id, comparison_id, run_id) "
            "where r.world_id=%s and r.comparison_id=%s group by 1, 2, 3 order by 1, 2, 3",
            (world_id, comparison_id),
        ).fetchall()
        drawn = connection.execute(
            "select count(distinct run_id) as n from society_comparison_replay "
            "where world_id=%s and comparison_id=%s",
            (world_id, comparison_id),
        ).fetchone()["n"]

    def utc(value: Any) -> str | None:
        return None if value is None else value.astimezone(dt.UTC).isoformat()

    return {
        "defined_utc": utc(defined),
        "first_outcome_utc": utc(span["first"]),
        "last_outcome_utc": utc(span["last"]),
        "asks": asks["asks"],
        "attempts": asks["attempts"],
        "runs": [dict(row) for row in statuses],
        "drawn_runs": drawn,
    }


def _write_record(
    state: Mapping[str, Any],
    name: str,
    comparison_id: str,
    *,
    notes: Mapping[str, Any],
    measured: Mapping[str, Any],
) -> dict[str, Any]:
    """Write the record of comparison ``comparison_id``, pre-registered as ``name``, from its
    stored runs: the result exactly as the application's route serves it, what the database
    recorded of its playing, its spend by model, the tree and both scripts by digest. ``notes``
    are statements the writer makes that the data cannot, printed as given; ``measured`` holds
    what the run step itself measured, integers and Decimal strings only."""
    paths = _paths(name)
    registered = json.loads((ROOT / paths["preregistration"]).read_text(encoding="utf-8"))
    town = registered["record"]["town"]
    served = _served(state, town["world_id"], town["version_id"], comparison_id)
    own = f"docs/evaluation/artifacts/{name}/record-as-run.py.txt"
    copy = ROOT / own
    copy.parent.mkdir(parents=True, exist_ok=True)
    if not copy.exists():
        copy.write_bytes(Path(__file__).read_bytes())
    record = {
        "comparison_id": comparison_id,
        "preregistration": paths["preregistration"],
        "preregistration_record_sha256": registered["record_sha256"],
        "verdict": served["verdict"],
        "result": served,
        "means_over": "each arm's mean_score is over its completed runs only; a failed run "
        "carries no score",
        "database": _database_facts(state, town["world_id"], comparison_id),
        "spend": _spend(state, town["world_id"], comparison_id),
        "bound_usd": registered["record"]["bound"]["bound_usd"],
        "measured": dict(measured),
        "notes": dict(notes),
        "tree": _tree(),
        "run_script_as_run": registered["record"].get("script_as_run", paths["script_as_run"]),
        "run_script_sha256": registered["record"]["script_sha256"],
        "record_script_as_run": own,
        "record_script_sha256": _sha256(Path(__file__).read_bytes()),
    }
    _write_new(paths["record"], _sealed(record, "exulanica.town-comparison/v1"))
    return record


def record(arguments: argparse.Namespace) -> None:
    state = json.loads(arguments.slot_state.read_text(encoding="utf-8"))
    notes = json.loads(arguments.notes.read_text(encoding="utf-8"))
    written = _write_record(state, arguments.name, arguments.comparison, notes=notes, measured={})
    print(json.dumps({"written": _paths(arguments.name)["record"], "verdict": written["verdict"]}))


def run(arguments: argparse.Namespace) -> None:
    from exulanica.api.services import build_services
    from exulanica.api.society_comparison_runner import ComparisonArm, RunHost
    from exulanica.api.society_comparison_start import comparison_cost
    from exulanica.orchestration.compare import comparison_body
    from exulanica.world.society_comparison_repository import seed_digest
    from exulanica.world.society_comparison_result import definition_role, protocol_value

    paths = _paths(arguments.name)
    if (ROOT / paths["record"]).exists():
        raise SystemExit(f"{paths['record']} exists: the comparison is judged once")
    registered = json.loads((ROOT / paths["preregistration"]).read_text(encoding="utf-8"))
    record_pre = registered["record"]
    if _tree() != record_pre["tree"]:
        raise SystemExit("the tree is not the pre-registered one")
    if _sha256(Path(__file__).read_bytes()) != record_pre["script_sha256"]:
        raise SystemExit("this script is not the pre-registered one")
    if _now() >= record_pre["run_window"]["starts_before_utc"]:
        raise SystemExit("the registered window to start the run has passed")
    as_run = ROOT / paths["script_as_run"]
    as_run.parent.mkdir(parents=True, exist_ok=True)
    if not as_run.exists():
        as_run.write_bytes(Path(__file__).read_bytes())
    state = json.loads(arguments.slot_state.read_text(encoding="utf-8"))
    bound = record_pre["bound"]["bound_usd"]
    ceiling = int(record_pre["process_calls"]["process_call_ceiling"])
    _environment(state, bound, arguments.dotenv, max_calls=ceiling)
    committed = [entry["seed_digest"] for entry in record_pre["held_out_seeds"]]
    found = {
        seed_digest(line.strip()): line.strip()
        for line in arguments.seeds.read_text(encoding="utf-8").splitlines()
        if line.strip() and not line.startswith("#")
    }
    if any(digest not in found for digest in committed):
        raise SystemExit("the seeds file does not hold every registered held-out seed")
    seeds = [found[digest] for digest in committed]
    services = build_services()
    budget = services.model_client.budget
    if budget.max_calls != ceiling or str(budget.ceiling_usd) != str(Decimal(bound)):
        raise SystemExit("the process's budget is not the registered bound and call ceiling")
    world_id = record_pre["town"]["world_id"]
    version = uuid.UUID(record_pre["town"]["version_id"])
    runner = services.comparison_runner(
        uuid.UUID(state["workspace_id"]), world_id, uuid.UUID(state["actor"])
    )
    if runner is None:
        raise SystemExit("this environment configures no society runtime")
    people = [person["id"] for person in record_pre["group"]["people"]]
    group = {"people": sorted(people), "source": {"kind": "named"}}
    others = runner.others_for(version, group["people"])
    body = comparison_body(
        runner,
        [ComparisonArm(provider, model_id) for provider, model_id in MODELS],
        seeds,
        control=True,
        phase="held_out",
        preregistration={
            "record": paths["preregistration"],
            "record_sha256": registered["record_sha256"],
        },
        group=group,
        others=others,
    )
    if body["claim"] != record_pre["claim"]:
        raise SystemExit("the definition's claim is not the registered one")
    comparison_id = uuid.uuid4()
    runner.define(version, comparison_id=comparison_id, body=body)
    if _frozen_input(state, world_id, str(comparison_id)) != record_pre["town"]["frozen_input"]:
        raise SystemExit("the comparison froze another input than the registered one")
    run_ids = runner.reserve_all(comparison_id, seeds)
    with services.database.session(uuid.UUID(state["workspace_id"])) as connection:
        definition = runner._repository(connection)._definition(comparison_id)["document"]
    role = definition_role(definition)

    def admit(runs: Sequence[uuid.UUID]) -> bool:
        with services.database.session(uuid.UUID(state["workspace_id"])) as connection:
            repository = runner._repository(connection)
            rows = [repository._run(run_id) for run_id in runs]
        seed = comparison_cost(
            definition,
            int(definition["population"]),
            role,
            budget,
            runner.manifest,
            at_once=protocol_value(runner.catalogs, "runs_at_once"),
            navigation_profile=record_pre["town"]["ground"].get("navigation_profile")
            or "city-walking-surfaces/v1",
            runs_left=[(row["arm"], row["seed_digest"]) for row in rows],
        )
        needed = seed.suggested_usd
        money = needed is None or budget.ceiling_usd - budget.spent_usd >= needed
        return money and budget.available_calls >= seed.calls

    host = RunHost(
        minute=lambda: None, stopping=lambda: False, recorded=lambda _c: None, admit=admit
    )
    started = time.monotonic_ns()
    runner.run_all(comparison_id, run_ids, host=host)
    ran_ms = (time.monotonic_ns() - started) // 1_000_000
    drawn = runner.draw_all(comparison_id, run_ids)
    api = _api(state)
    route = f"/world/versions/{version}/society/comparisons/{comparison_id}"
    read_ms = []
    result_served = api(route, [("world_id", world_id)])
    for seed in result_served["seeds"][:2]:
        for arm in ("model_a", "model_b"):
            if seed["runs"][arm]["status"] != "completed":
                continue
            began = time.monotonic_ns()
            api(f"{route}/runs/{seed['runs'][arm]['run_id']}", [("world_id", world_id)])
            read_ms.append((time.monotonic_ns() - began) // 1_000_000)
    written = _write_record(
        state,
        arguments.name,
        str(comparison_id),
        notes={},
        measured={
            "run_ms": ran_ms,
            "drawn_by_this_process": drawn,
            "stored_drawing_read_ms": read_ms,
            "spent_usd": str(budget.spent_usd),
            "billed_calls": budget.billed_calls,
            "process_call_ceiling": budget.max_calls,
        },
    )
    print(json.dumps({"written": paths["record"], "verdict": written["verdict"]}))


def main(argv: Sequence[str] | None = None) -> None:
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("step", choices=("preregister", "run", "record", "rewrap"))
    parser.add_argument("--path", help="for rewrap: the record's path under the repository")
    parser.add_argument(
        "--name", required=False, help="the records' name, e.g. <date>-town-comparison"
    )
    parser.add_argument("--comparison")
    parser.add_argument("--notes", type=Path)
    parser.add_argument("--predecessor")
    parser.add_argument("--predecessor-record")
    parser.add_argument("--slot-state", type=Path, required=True)
    parser.add_argument("--world")
    parser.add_argument("--version", type=uuid.UUID)
    parser.add_argument("--control", action="append")
    parser.add_argument("--bound-usd")
    parser.add_argument("--seeds", type=Path)
    parser.add_argument("--dotenv", type=Path, help="the operator's .env, read for the key alone")
    arguments = parser.parse_args(argv)
    if arguments.step == "preregister":
        preregister(arguments)
    elif arguments.step == "record":
        record(arguments)
    elif arguments.step == "rewrap":
        rewrap(arguments)
    else:
        run(arguments)


if __name__ == "__main__":
    main(sys.argv[1:])
