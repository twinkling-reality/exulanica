"""What one slow flight request does to the requests beside it, on a running server.

    python3 measure_request_blocking.py --state RUN_DIR/state.json --setup SETUP_JSON --out FILE

The flight route runs the flight module in Python, which holds the interpreter's lock while it
computes. This asks the server for GET /healthz every 20 ms, first alone and then while one cold
window at the step before an episode ends (the most a window computes) is being computed, and
records how long each health answer took and how long the window took. It repeats with a different
episode each time, so no window is resumed from a cached state.

The token is read from the launcher's token file and sent only to the local server; it is never
printed or written. Every location comes from the arguments or the launcher's state file.
"""

import argparse
import json
import threading
import time
import urllib.parse
import urllib.request
from pathlib import Path

EPISODE = 3000
INTERVAL_S = 0.020
REPEATS = 5


def _get(url, token):
    started = time.perf_counter()
    request = urllib.request.Request(url, headers={"Authorization": " ".join(("Bearer", token))})
    with urllib.request.urlopen(request, timeout=60) as response:
        response.read()
        status = response.status
    return status, round((time.perf_counter() - started) * 1_000_000)


def _quantile(values, q):
    ordered = sorted(values)
    return ordered[min(len(ordered) - 1, max(0, -(-len(ordered) * q // 1) - 1))] if ordered else None


def _summary(values):
    return {
        "count": len(values),
        "p50_us": _quantile(values, 0.5),
        "p95_us": _quantile(values, 0.95),
        "max_us": max(values) if values else None,
    }


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--state", required=True)
    parser.add_argument("--setup", required=True)
    parser.add_argument("--out", required=True)
    arguments = parser.parse_args()
    state = json.loads(Path(arguments.state).read_text())
    setup = json.loads(Path(arguments.setup).read_text())
    token = Path(state["token_file"]).read_text().strip()
    base = f"http://127.0.0.1:{state['ports']['api']}"
    health = base + "/healthz"
    scope = urllib.parse.urlencode({"world_id": setup["world_id"]})
    runs = []
    for repeat in range(REPEATS):
        alone = []
        for _ in range(50):
            alone.append(_get(health, token)[1])
            time.sleep(INTERVAL_S)
        from_step = (288 - repeat) * EPISODE - 1
        flight_url = (
            f"{base}/world/versions/{setup['version_id']}/flight?{scope}&from_step={from_step}&steps=600"
        )
        answer = {}

        def fly(url=flight_url, answer=answer):
            answer["status"], answer["us"] = _get(url, token)

        thread = threading.Thread(target=fly)
        beside = []
        thread.start()
        while thread.is_alive():
            beside.append(_get(health, token)[1])
            time.sleep(INTERVAL_S)
        thread.join()
        runs.append(
            {
                "from_step": from_step,
                "flight_status": answer["status"],
                "flight_us": answer["us"],
                "health_alone": _summary(alone),
                "health_beside_the_flight": _summary(beside),
            }
        )
    record = {
        "profile": "exulanica.flight-bounds/v2/request-blocking",
        "flyers": setup["flyers"],
        "interval_ms": round(INTERVAL_S * 1000),
        "runs": runs,
    }
    Path(arguments.out).write_text(json.dumps(record, indent=1))
    print(json.dumps({"runs": [[run["flight_us"], run["health_beside_the_flight"]["max_us"]] for run in runs]}))


if __name__ == "__main__":
    main()
