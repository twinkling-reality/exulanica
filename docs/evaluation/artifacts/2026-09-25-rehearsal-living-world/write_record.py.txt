"""Write the digest-bound evaluation record of one rehearsal run on main, with its as-run driver.

Lane file, published beside the record's artifacts as write_record.py.txt. Run it from the rehearsal
tree's root with the product's environment, which supplies the canonical JSON the record's digest is
taken over and the phrases the retained-records test refuses:

    .venv/bin/python <this file> --run RUN_DIR --name NAME --earlier RECORD_JSON

Every fact in the record is read from what the rehearsal wrote in RUN_DIR (result.json, summary.txt,
screens/, the session logs) or computed from bytes. Machine locations are found at run time (the
home directory, the temporary directory, the account name, this tree, the main checkout, the paths
the run was started with) and replaced by placeholders, and nothing is written while any of them, or
the run's own token, is still in a published text.

The as-run driver is every file of scripts/rehearsal as the run recorded it: the run wrote the
SHA-256 of each file that ran (run.rehearsal_files), and each file is published only if the tree's
bytes still hash to that digest, so the published copies are the bytes that ran.
"""

LOAD_LINE = re.compile(r"^gpu-slot: (?:slot taken|finished) at .*; load (?P<one>[0-9.]+) ")


def sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def session_load(run: Path) -> dict[str, str]:
    """The one-minute load average the GPU slot logged as each browser session began and ended."""
    loads = [
        Decimal(match.group("one"))
        for log in sorted(run.glob("sessions/*/session.log"))
        for match in (LOAD_LINE.match(line) for line in log.read_text().splitlines())
        if match is not None
    ]
    assert loads, "no session log records the machine's load"
    return {"one_minute_min": str(min(loads)), "one_minute_max": str(max(loads)), "readings": str(len(loads))}


def argument_after(argv: list[str], flag: str) -> str | None:
    return argv[argv.index(flag) + 1] if flag in argv else None


def scrubber(result: dict) -> tuple[list[tuple[str, str]], list[str], list[str]]:
    """The replacements that remove every machine location; the machine fragments no published byte
    may hold; and what no published record or run output may hold (the placeholder the run itself
    writes for the home directory, and the phrases the retained-records test refuses)."""
    home = str(Path.home())
    homed = lambda path: str(path).replace(home, "<home>")  # noqa: E731 - both spellings are needed
    command = result["run"]["command"]
    pairs = []
    for flag, placeholder in (("--worktree", "<application-tree>"), ("--model-env", "<model-environment-file>"), ("--out", "<run>")):
        value = argument_after(command, flag)
        # A relative argument (".") names no machine location, and replacing it would rewrite text.
        if value is not None and (value.startswith("/") or value.startswith("<home>")):
            pairs += [(value, placeholder), (homed(value), placeholder)]
    for location, placeholder in ((MAIN_CHECKOUT, "<main-checkout>"), (TREE, "<rehearsal-tree>")):
        pairs += [(str(location), placeholder), (homed(location), placeholder)]
    temporary = tempfile.gettempdir()
    for location in {temporary, os.path.realpath(temporary)}:
        pairs.append((location.rstrip(os.sep), "<temporary>"))
    # Longest first, so a path is replaced whole before any prefix of it.
    pairs.sort(key=lambda pair: len(pair[0]), reverse=True)
    machine = [home, str(Path.home().parent) + os.sep, temporary, os.path.realpath(temporary), getpass.getuser()]
    outputs = [homed(Path.home()), *visual_gate._FORBIDDEN_TEXT]
    return pairs, machine, outputs


def quantised(value: object, key: str = "") -> object:
    """The value with every duration in whole milliseconds; the record's digest refuses floats."""
    if isinstance(value, dict):
        out = {}
        for name, item in value.items():
            if (name == "seconds" or name.endswith("_seconds")) and isinstance(item, float):
                out[name.removesuffix("seconds") + "milliseconds"] = round(item * 1000)
            else:
                out[name] = quantised(item, name)
        return out
    if isinstance(value, list):
        return [quantised(item, key) for item in value]
    if isinstance(value, float):
        raise ValueError(f"a fractional {key!r} would enter the record's digest; quantise it by name")
    return value


def decimal_strings(value: object) -> object:
    """The value with every fraction written as a decimal string; the record's digest refuses floats."""
    if isinstance(value, dict):
        return {name: decimal_strings(item) for name, item in value.items()}
    if isinstance(value, list):
        return [decimal_strings(item) for item in value]
    if isinstance(value, float):
        return repr(value)
    return value


def living_world(by_id: dict) -> dict:
    """What the living session measured and read, from the observations the run recorded."""
    def observed(step: str, observation: str) -> object:
        found = [o for o in by_id[step].get("observations", []) if o["id"] == observation]
        return found[0]["observed"] if found else None
    motion = observed("play-and-watch-walking", "walking-continuously") or {}
    return decimal_strings({
        "statuses": {step: by_id[step]["status"] for step in (
            "start-with-small-square", "furnish-small-square", "bring-in-inhabitants", "play-and-watch-walking",
            "inspect-an-inhabitant", "object-in-use", "pause-playback")},
        "measure": "lane W2's: per animation frame, the ground displacement of every outdoor walker the page draws in two consecutive frames; judged by the 95th percentile pace of the walker-frames at or above the walking floor; the moving share (walker-frames with any displacement) is reported, not judged",
        "motion": {key: motion.get(key) for key in ("seconds", "frames", "walkers_seen", "frame_ms", "walker_frames",
                                                    "moving_walker_frames", "moving_share", "step_m", "speed_bands", "walking",
                                                    "pace_p95_at_most_m_per_s")},
        "per_walker": motion.get("walkers"),
        "host": observed("play-and-watch-walking", "host-plays"),
        "minutes": observed("play-and-watch-walking", "minutes-pass"),
        "inspector": observed("inspect-an-inhabitant", "who-what-why"),
        "in_use": observed("object-in-use", "use-recorded"),
        "paused": observed("pause-playback", "minutes-stop"),
    })


def git_out(*args: str) -> str:
    return subprocess.run(["git", *args], cwd=TREE, capture_output=True, text=True, check=True).stdout.strip()


def composition(arguments, tree: dict) -> dict:
    """What the application tree was: main at a commit, one held commit's diff and the rehearsal's
    delta applied on it, both unstaged, and the commits main gained since the earlier record."""
    assert tree["head"] == git_out("rev-parse", arguments.application_main), "the tree's head is not the named main"
    applied = git_out("log", "-1", "--format=%H %s", arguments.applied).split(" ", 1)
    since = [line.split(" ", 1) for line in git_out("log", "--format=%H %s", f"{arguments.since}..{arguments.application_main}").splitlines()]
    return {
        "main": tree["head"],
        "applied_unstaged": [
            {"commit": applied[0], "subject": applied[1], "relation": "held for landing with the rehearsal's delta; documents only"},
            {"delta_patch_sha256": arguments.delta_sha256, "relation": "the rehearsal's own change: step list, drivers, tests and section 5 of docs/demo-integrity.md"},
        ],
        "tree_diff_head_sha256": tree["diff_head_sha256"],
        "main_commits_since_earlier_record": [{"commit": c, "subject": s} for c, s in since],
    }


def gates_read_from(applied: str) -> dict:
    """The document the gates were read from, by the bytes in the tree and the commit that wrote them."""
    data = (TREE / "docs" / "product-direction.md").read_bytes()
    held = subprocess.run(["git", "show", f"{applied}:docs/product-direction.md"], cwd=TREE, capture_output=True, check=True).stdout
    assert data == held, "the tree's product-direction.md is not the applied commit's"
    return {"document": "docs/product-direction.md", "sha256": sha256(data), "written_by_commit": git_out("rev-parse", applied)}


def scrub(text: str, pairs: list[tuple[str, str]]) -> str:
    for local, placeholder in pairs:
        text = text.replace(local, placeholder)
    return text


def run_token(result: dict) -> str | None:
    """The run's synthetic token, from the launcher's kept run record, to prove it is published nowhere."""
    spec = importlib.util.spec_from_file_location("acceptance_launch", TREE / "scripts" / "acceptance" / "launch.py")
    launcher = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(launcher)
    workspace = result["run"]["runtime"]["workspace_id"]
    for state in launcher.runtime_root().glob("*/state.json"):
        kept = json.loads(state.read_text())
        if kept.get("workspace_id") == workspace and Path(kept["token_file"]).exists():
            return Path(kept["token_file"]).read_text().strip()
    for state in launcher.runtime_root().glob("*/*/state.json"):
        kept = json.loads(state.read_text())
        if kept.get("workspace_id") == workspace and Path(kept["token_file"]).exists():
            return Path(kept["token_file"]).read_text().strip()
    return None


def as_run_driver(result: dict) -> tuple[dict, dict[str, bytes]]:
    """Every rehearsal file the run recorded, checked against its recorded digest, to publish."""
    recorded = result["run"]["rehearsal_files"]
    files: dict[str, dict] = {}
    published: dict[str, bytes] = {}
    present = sorted(p.name for p in REHEARSAL.iterdir() if p.is_file())
    assert present == sorted(recorded), f"the tree's rehearsal files {present} are not the run's {sorted(recorded)}"
    for name, digest in sorted(recorded.items()):
        data = (REHEARSAL / name).read_bytes()
        assert sha256(data) == digest, f"{name} differs from the bytes the run recorded"
        published_as = f"driver/{name}.txt"
        files[name] = {"sha256": digest, "published_as": published_as}
        published[published_as] = data
    return files, published


def predecessor(path: Path) -> dict:
    """The earlier rehearsal record of main, bound by the digest its own record reproduces."""
    document = json.loads(path.read_bytes())
    digest = hashlib.sha256(canonical_json(document["record"])).hexdigest()
    assert digest == document["record_sha256"], f"{path.name} does not reproduce its digest"
    return {"path": str(path.relative_to(TREE)), "record_sha256": digest,
            "relation": "the earlier recorded rehearsal of main, which walked the personal path and ended at a made world's refusal of a new photograph"}


def earlier_record(path: Path, result: dict) -> dict:
    """What the earlier published rehearsal record of main reported, step by step, beside this run."""
    raw = path.read_bytes()
    earlier = json.loads(raw)["record"]
    before = {s["id"]: s for s in earlier["steps"]}
    now = {s["id"]: s for s in result["steps"]}
    return {
        "application_head": earlier["application_tree"]["head"],
        "summary": earlier["summary"],
        "steps_only_there": sorted(set(before) - set(now)),
        "steps_only_here": sorted(set(now) - set(before)),
        "differing_steps": [
            {"step": step_id, "earlier_status": before[step_id]["status"], "this_status": now[step_id]["status"]}
            for step_id in before
            if step_id in now and before[step_id]["status"] != now[step_id]["status"]
        ],
        # The gates each step serves as docs/product-direction.md stated them for each run.
        "gates_moved": [
            {"step": step_id, "earlier_gates": before[step_id]["gates"], "these_gates": now[step_id]["gates"]}
            for step_id in before
            if step_id in now and before[step_id]["gates"] != now[step_id]["gates"]
        ],
    }


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run", required=True)
    parser.add_argument("--name", required=True)
    parser.add_argument("--earlier", required=True)
    parser.add_argument("--application-main", required=True, help="the main commit the application tree was made at")
    parser.add_argument("--applied", required=True, help="a commit not on main whose diff the tree holds unstaged")
    parser.add_argument("--delta-sha256", required=True, help="the SHA-256 of the rehearsal's delta.patch the tree holds unstaged")
    parser.add_argument("--since", required=True, help="the main commit the earlier record's application ran at")
    arguments = parser.parse_args()
    run = Path(arguments.run).resolve()
    raw = (run / "result.json").read_bytes()
    result = json.loads(raw)
    pairs, machine, outputs = scrubber(result)
    token = run_token(result)
    assert token is not None, "the run's token was not found, so its absence cannot be checked"
    machine.append(token)

    artifacts_dir = TREE / "docs" / "evaluation" / "artifacts" / arguments.name
    relative = f"docs/evaluation/artifacts/{arguments.name}"
    if artifacts_dir.exists():
        shutil.rmtree(artifacts_dir)
    (artifacts_dir / "driver").mkdir(parents=True)
    texts: dict[str, bytes] = {
        "result.json": scrub(raw.decode(), pairs).encode(),
        "summary.txt": scrub((run / "summary.txt").read_text(), pairs).encode(),
        "write_record.py.txt": Path(__file__).read_bytes(),
    }
    files, driver = as_run_driver(result)
    texts.update(driver)
    for name, data in texts.items():
        # Source copies may hold the placeholder and the refused phrases as code; data may not.
        checked = machine if name.endswith(".txt") and name != "summary.txt" else machine + outputs
        for fragment in checked:
            assert fragment.encode() not in data, f"{name} would publish a machine location or credential"
        (artifacts_dir / name).write_bytes(data)

    by_id = {step["id"]: step for step in result["steps"]}
    captures = []
    for step_id, label, shows in CAPTURES:
        shots = [s for s in by_id[step_id].get("evidence", {}).get("screenshots", []) if s["file"].endswith(f"-{step_id}-{label}.png")]
        if not shots:
            continue
        [shot] = shots
        data = (run / shot["file"]).read_bytes()
        assert sha256(data) == shot["sha256"], f"{shot['file']} is not the image the run recorded"
        target = artifacts_dir / Path(shot["file"]).name
        target.write_bytes(data)
        captures.append({"path": f"{relative}/{target.name}", "byte_size": len(data), "sha256": sha256(data),
                         "step": step_id, "shows": shows})

    def bound(name: str) -> dict:
        data = (artifacts_dir / name).read_bytes()
        return {"path": f"{relative}/{name}", "byte_size": len(data), "sha256": sha256(data)}

    kept = {key: value for key, value in result["run"].items() if key != "facts"}
    run_block = quantised(json.loads(scrub(json.dumps(kept), pairs)))
    rights = by_id["grant-model-rights-in-app"]
    steps_list = json.loads((REHEARSAL / "steps.json").read_bytes())
    stand_ins = {
        key: {**entry, "given_by_steps": [s["id"] for s in steps_list["steps"] if s.get("stand_in") == key],
              "resting_steps": [s["id"] for s in result["steps"] if any(x["key"] == key for x in s.get("stand_ins", []))],
              "gates": [{"gate": g["gate"], "status": g["status"]} for g in result["gates"] if key in g["stand_ins"]]}
        for key, entry in steps_list["stand_ins"].items()
    }
    record = {
        "profile": "exulanica.rehearsal-run/v1",
        "scope": (
            "One run of the automated rehearsal of the first demonstration (scripts/rehearsal, step list "
            "version 6) against main in the real application: a production build in headless Chrome over the DevTools "
            "protocol, one page per session, a fresh synthetic workspace on acceptance slot 3, and the "
            "hosted models through the product's configured client, with main's production derivative "
            "worker and its depth model, and the host playing the run's workspace at its shipped pace. "
            "The living world runs in the starter: the first-use card's small square placed and taken "
            "back, the Create panel's square, inhabitants brought in, the world played and watched "
            "walking, an inhabitant inspected, someone using the square, and a pause. The person's path runs from uploading synthetic photographs in the "
            "drawer through their admission, model rights and review to the depth pass and scene grouping, "
            "a world made from them, the Companion asked inside it, and the return to it, where one more "
            "photograph is added to the made world on a confirmed preview. The human review is "
            "the rehearsal driver's stand-in for a person's review, stated in stand_in_disclosure; no person "
            "reviewed these photographs. Each step is a person's action with its declared observable results; "
            "each gate of docs/product-direction.md is reported with its first failed step, and a gate that "
            "rests on the stand-in never as a plain pass."
        ),
        "corpus_class": "synthetic",
        "corpus": {
            "workspace": run_block["runtime"]["workspace"],
            "photographs": {session: inputs["photographs"] for session, inputs in run_block["prepared"].items()},
            "photographs_reason": {session: inputs["photographs_reason"] for session, inputs in run_block["prepared"].items()},
            "personal_media_admitted": False,
        },
        "stand_in_disclosure": stand_ins,
        "application_tree": run_block["application_tree"] | {"main_at_run": arguments.application_main},
        "rehearsal": {
            "steps_profile": result["steps_profile"],
            "steps_version": result["steps_version"],
            "steps_sha256_as_run": result["run"]["steps_sha256"],
            "rehearsal_tree": run_block["rehearsal_tree"],
            "driver": files,
            "driver_established": "each file's bytes hash to the SHA-256 the run recorded for it in run.rehearsal_files",
        },
        "runtime": run_block["runtime"],
        "build": run_block["build"],
        "access": run_block["access"],
        "model": run_block["model"],
        "browser": run_block["browser"],
        "summary": result["summary"],
        "steps": [
            {"id": s["id"], "status": s["status"], "gates": s["gates"], "session": s.get("session"),
             "owner_area": s["owner_area"], "reason": s.get("reason"), "spend_usd": s.get("spend_usd"),
             "stand_ins": [x["key"] for x in s.get("stand_ins", [])],
             "observations": [{"id": o["id"], "ok": o["ok"]} for o in s.get("observations", [])]}
            for s in result["steps"]
        ],
        "gates": result["gates"],
        "failures": decimal_strings([
            {"step": s["id"], "reason": s["reason"], "observations": s["observations"]}
            for s in result["steps"] if s["status"] == "failed"
        ]),
        "living_world": living_world(by_id),
        "application_composition": composition(arguments, run_block["application_tree"]),
        "gates_read_from": gates_read_from(arguments.applied),
        "model_rights_step": quantised({
            "status": rights["status"],
            "observations": rights.get("observations", []),
        }),
        "spend": {
            **run_block["model"]["spend"],
            "by_step": {s["id"]: s["spend_usd"] for s in result["steps"] if s.get("spend_usd") is not None},
        },
        "predecessor_record": predecessor(Path(arguments.earlier).resolve()),
        "compared_with_predecessor": earlier_record(Path(arguments.earlier).resolve(), result),
        "captures": captures,
        "artifacts": [bound(name) for name in sorted(texts)],
        "result_sha256_as_written_by_the_run": sha256(raw),
        "scrubbing": (
            "machine locations in the published result and summary are replaced by <application-tree>, "
            "<model-environment-file>, <run>, <main-checkout>, <rehearsal-tree> and <temporary>, and "
            "neither holds the placeholder the run wrote for the home directory. The images are the bytes "
            "the run captured, and each matches the SHA-256 the run recorded for it. Durations the run "
            "wrote in seconds are given here in whole milliseconds."
        ),
        "what_this_does_not_show": [
            "The human review of every photograph was recorded by the rehearsal driver through the drawer's own controls as a stand-in for a person's review; no person inspected these photographs, and every gate resting on it reports passed_on_stand_in or worse, never a plain pass.",
            "The photographs are synthetic drawings in a synthetic workspace; no personal photograph is involved.",
            "The depth pass ran MoGe-2 on this machine's CPU in main's production derivative worker. No multi-view reconstruction ran: no scene group held three photographs, and the scene worker was not started (reconstruct-scene-geometry states why).",
            "The proposed place was confirmed through the identity routes rather than in the application.",
            "Stopping a right in the drawer was not exercised by this run.",
            "One run: the Companion's answers vary between runs.",
            "Access used the production build's operator token gate; account sign-in was not exercised.",
            "No person observed the run and no demonstration footage was recorded; no frame time was judged. The walking measure counts the frames in which drawn walkers moved, over one window of one run.",
            "The living world was watched in the starter at the host's shipped pace with one population of eight; the walking threshold comes from lane W2's two measured arms, not from a pre-registered held-out set.",
            "Other work loaded the machine during the run (machine_load, from the GPU slot's own log lines); the times are not timing measurements.",
        ],
        "machine_load": session_load(run),
        "completed_at": result["run"]["finished_at"],
    }
    text = json.dumps(
        {"profile": "exulanica.digest-bound-record/v1", "record": record,
         "record_sha256": hashlib.sha256(canonical_json(record)).hexdigest()},
        indent=2, sort_keys=True,
    ) + "\n"
    for fragment in machine + outputs:
        assert fragment not in text, "the record would publish a machine location or credential"
    target = TREE / "docs" / "evaluation" / f"{arguments.name}.json"
    target.write_text(text)
    print(f"wrote {target.relative_to(TREE)}: {len(captures)} captures, {len(record['artifacts'])} artifacts, token checked")
    return 0


if __name__ == "__main__":
    sys.exit(main())
