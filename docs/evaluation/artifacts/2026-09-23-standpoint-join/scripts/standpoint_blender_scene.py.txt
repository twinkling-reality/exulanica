"""Build one procedural scene from a JSON spec inside Blender and render every camera it names.

Run by standpoint_synthetic_sets.py, never by hand:

    Blender --background --factory-startup --python-exit-code 1 \
        --python standpoint_blender_scene.py -- <spec.json> <out_dir>

Everything random was decided by the driver and written into the spec, so this file makes no
choice of its own: the spec IS the scene. For each camera it writes the photograph as an 8-bit
sRGB PNG and the ground-truth depth as a little-endian float32 array (<id>.depth.f32, row-major,
top row first), then one manifest describing what it wrote.

Depth is the Cycles Z pass, written as a float EXR by the compositor and read back here. It is
planar distance along the camera's view axis: MEASURED on a probe scene, a wall facing the camera
at 9.9 m read 9.9 across a whole row, where distance to the camera centre would grow at the edges.
Spec colours are sRGB values and are converted to Blender's linear space on the way in.
"""

import json
import math
import sys
import time
from pathlib import Path

import bpy
from mathutils import Euler, Vector

ARGV = sys.argv[sys.argv.index("--") + 1 :]
SPEC_PATH = Path(ARGV[0])
OUT_DIR = Path(ARGV[1])
SPEC = json.loads(SPEC_PATH.read_text(encoding="utf-8"))


def srgb(color):
    """Spec colours are sRGB values, as a person would pick them; Blender's nodes are linear."""
    return tuple(
        (c / 12.92) if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4 for c in color
    ) + (1.0,)


def _principled(material):
    nodes = material.node_tree.nodes
    return nodes.get("Principled BSDF")


def _ramp(nodes, links, source_socket, colors, positions):
    ramp = nodes.new("ShaderNodeValToRGB")
    elements = ramp.color_ramp.elements
    while len(elements) < len(colors):
        elements.new(0.5)
    for element, color, position in zip(elements, colors, positions):
        element.position = position
        element.color = srgb(color)
    links.new(source_socket, ramp.inputs["Fac"])
    return ramp


def _mapping(nodes, links, scale, seed):
    coordinates = nodes.new("ShaderNodeTexCoord")
    mapping = nodes.new("ShaderNodeMapping")
    mapping.inputs["Scale"].default_value = (scale, scale, scale)
    # A per-material offset so two materials of one kind do not show the same pattern.
    mapping.inputs["Location"].default_value = (seed * 7.31, seed * 3.17, seed * 5.13)
    links.new(coordinates.outputs["Object"], mapping.inputs["Vector"])
    return mapping


def _bump(nodes, links, height_socket, strength, bsdf):
    bump = nodes.new("ShaderNodeBump")
    bump.inputs["Strength"].default_value = strength
    links.new(height_socket, bump.inputs["Height"])
    links.new(bump.outputs["Normal"], bsdf.inputs["Normal"])


def material_brick(material, spec, nodes, links, bsdf):
    mapping = _mapping(nodes, links, spec["scale"], spec["seed"])
    brick = nodes.new("ShaderNodeTexBrick")
    brick.inputs["Scale"].default_value = spec.get("brick_scale", 1.0)
    brick.inputs["Color1"].default_value = srgb(spec["color"])
    brick.inputs["Color2"].default_value = srgb(spec["color2"])
    brick.inputs["Mortar"].default_value = srgb(spec["mortar"])
    brick.inputs["Mortar Size"].default_value = spec.get("mortar_size", 0.02)
    brick.inputs["Brick Width"].default_value = spec.get("brick_width", 0.5)
    brick.inputs["Row Height"].default_value = spec.get("row_height", 0.25)
    links.new(mapping.outputs["Vector"], brick.inputs["Vector"])
    noise = nodes.new("ShaderNodeTexNoise")
    noise.inputs["Scale"].default_value = 40.0
    noise.inputs["Detail"].default_value = 8.0
    links.new(mapping.outputs["Vector"], noise.inputs["Vector"])
    mix = nodes.new("ShaderNodeMixRGB")
    mix.blend_type = "MULTIPLY"
    mix.inputs["Fac"].default_value = 0.35
    links.new(brick.outputs["Color"], mix.inputs["Color1"])
    links.new(noise.outputs["Color"], mix.inputs["Color2"])
    links.new(mix.outputs["Color"], bsdf.inputs["Base Color"])
    _bump(nodes, links, brick.outputs["Fac"], 0.4, bsdf)


def material_noise(material, spec, nodes, links, bsdf):
    """Plaster, grass, foliage, fabric, rock and paint: a noise field through a colour ramp."""
    mapping = _mapping(nodes, links, spec["scale"], spec["seed"])
    noise = nodes.new("ShaderNodeTexNoise")
    noise.inputs["Scale"].default_value = spec.get("noise_scale", 6.0)
    noise.inputs["Detail"].default_value = spec.get("detail", 10.0)
    noise.inputs["Roughness"].default_value = spec.get("roughness", 0.6)
    links.new(mapping.outputs["Vector"], noise.inputs["Vector"])
    colors = spec["colors"]
    positions = [index / max(1, len(colors) - 1) for index in range(len(colors))]
    ramp = _ramp(nodes, links, noise.outputs["Fac"], colors, [0.3 + 0.4 * p for p in positions])
    if spec.get("voronoi", False):
        # Cell structure as brightness only: the Voronoi "Color" output is a random hue per cell,
        # which drew rocks and trees as stained glass on the first trial renders.
        voronoi = nodes.new("ShaderNodeTexVoronoi")
        voronoi.inputs["Scale"].default_value = spec.get("voronoi_scale", 12.0)
        links.new(mapping.outputs["Vector"], voronoi.inputs["Vector"])
        cells = _ramp(
            nodes, links, voronoi.outputs["Distance"], [(1.0, 1.0, 1.0), (0.55, 0.55, 0.55)], [0.0, 0.9]
        )
        mix = nodes.new("ShaderNodeMixRGB")
        mix.blend_type = "MULTIPLY"
        mix.inputs["Fac"].default_value = 0.7
        links.new(ramp.outputs["Color"], mix.inputs["Color1"])
        links.new(cells.outputs["Color"], mix.inputs["Color2"])
        links.new(mix.outputs["Color"], bsdf.inputs["Base Color"])
    else:
        links.new(ramp.outputs["Color"], bsdf.inputs["Base Color"])
    _bump(nodes, links, noise.outputs["Fac"], spec.get("bump", 0.25), bsdf)


def material_wood(material, spec, nodes, links, bsdf):
    mapping = _mapping(nodes, links, spec["scale"], spec["seed"])
    wave = nodes.new("ShaderNodeTexWave")
    wave.inputs["Scale"].default_value = spec.get("wave_scale", 3.0)
    wave.inputs["Distortion"].default_value = 6.0
    wave.inputs["Detail"].default_value = 4.0
    links.new(mapping.outputs["Vector"], wave.inputs["Vector"])
    ramp = _ramp(nodes, links, wave.outputs["Fac"], spec["colors"], [0.2, 0.8])
    planks = nodes.new("ShaderNodeTexBrick")
    planks.inputs["Color1"].default_value = (1.0, 1.0, 1.0, 1.0)
    planks.inputs["Color2"].default_value = (0.8, 0.8, 0.8, 1.0)
    planks.inputs["Mortar"].default_value = (0.25, 0.25, 0.25, 1.0)
    planks.inputs["Mortar Size"].default_value = 0.01
    planks.inputs["Brick Width"].default_value = 2.0
    planks.inputs["Row Height"].default_value = 0.2
    links.new(mapping.outputs["Vector"], planks.inputs["Vector"])
    mix = nodes.new("ShaderNodeMixRGB")
    mix.blend_type = "MULTIPLY"
    mix.inputs["Fac"].default_value = 1.0
    links.new(ramp.outputs["Color"], mix.inputs["Color1"])
    links.new(planks.outputs["Color"], mix.inputs["Color2"])
    links.new(mix.outputs["Color"], bsdf.inputs["Base Color"])
    _bump(nodes, links, planks.outputs["Fac"], 0.2, bsdf)


def material_flat(material, spec, nodes, links, bsdf):
    bsdf.inputs["Base Color"].default_value = srgb(spec["color"])


def material_glass(material, spec, nodes, links, bsdf):
    bsdf.inputs["Base Color"].default_value = srgb(spec["color"])
    bsdf.inputs["Roughness"].default_value = 0.08
    bsdf.inputs["Metallic"].default_value = 0.6


def material_emission(material, spec, nodes, links, bsdf):
    bsdf.inputs["Emission Color"].default_value = srgb(spec["color"])
    bsdf.inputs["Emission Strength"].default_value = spec.get("strength", 4.0)


#: The material kinds a spec may name. A kind missing here is refused, never guessed.
MATERIALS = {
    "brick": material_brick,
    "noise": material_noise,
    "wood": material_wood,
    "flat": material_flat,
    "glass": material_glass,
    "emission": material_emission,
}


def make_material(name, spec):
    kind = spec["kind"]
    if kind not in MATERIALS:
        raise ValueError(f"no material recipe named {kind!r}")
    material = bpy.data.materials.new(name)
    material.use_nodes = True
    nodes = material.node_tree.nodes
    links = material.node_tree.links
    bsdf = _principled(material)
    bsdf.inputs["Roughness"].default_value = spec.get("surface_roughness", 0.7)
    MATERIALS[kind](material, spec, nodes, links, bsdf)
    return material


def add_shape(item):
    shape = item["shape"]
    location = Vector(item["location"])
    if shape == "box":
        bpy.ops.mesh.primitive_cube_add(size=1.0, location=location)
        obj = bpy.context.active_object
        obj.scale = Vector(item["size"])
    elif shape == "cylinder":
        bpy.ops.mesh.primitive_cylinder_add(
            radius=item["radius"], depth=item["height"], location=location, vertices=32
        )
        obj = bpy.context.active_object
    elif shape == "sphere":
        bpy.ops.mesh.primitive_ico_sphere_add(
            radius=item["radius"], subdivisions=4, location=location
        )
        obj = bpy.context.active_object
        obj.scale = Vector(item.get("squash", [1.0, 1.0, 1.0]))
    elif shape == "cone":
        bpy.ops.mesh.primitive_cone_add(
            radius1=item["radius"], depth=item["height"], location=location, vertices=48
        )
        obj = bpy.context.active_object
    elif shape == "plane":
        bpy.ops.mesh.primitive_plane_add(size=1.0, location=location)
        obj = bpy.context.active_object
        obj.scale = Vector((item["size"][0], item["size"][1], 1.0))
    elif shape == "heightfield":
        # A grid of heights the driver computed, so the ground under a standpoint is known
        # exactly rather than read back from a displacement modifier.
        count = item["count"]
        spacing = item["spacing"]
        heights = item["heights"]
        half = (count - 1) * spacing / 2
        vertices = [
            (column * spacing - half, row * spacing - half, heights[row * count + column])
            for row in range(count)
            for column in range(count)
        ]
        faces = [
            (
                row * count + column,
                row * count + column + 1,
                (row + 1) * count + column + 1,
                (row + 1) * count + column,
            )
            for row in range(count - 1)
            for column in range(count - 1)
        ]
        mesh = bpy.data.meshes.new(f"{item['id']}-mesh")
        mesh.from_pydata(vertices, [], faces)
        mesh.update()
        obj = bpy.data.objects.new(item["id"], mesh)
        bpy.context.collection.objects.link(obj)
        obj.location = location
        bpy.context.view_layer.objects.active = obj
        obj.select_set(True)
    else:
        raise ValueError(f"no shape named {shape!r}")
    obj.name = item["id"]
    rotation = item.get("rotation_deg", [0.0, 0.0, 0.0])
    obj.rotation_euler = Euler([math.radians(value) for value in rotation], "XYZ")
    # Texture coordinates are the object's own; applying scale makes them metres, so a pattern's
    # feature size means the same thing on a crate and on a wall.
    bpy.ops.object.select_all(action="DESELECT")
    obj.select_set(True)
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    if shape in ("sphere", "heightfield"):
        bpy.ops.object.shade_smooth()
    obj.data.materials.append(make_material(f"{item['id']}-material", item["material"]))
    return obj


def build_world(sky):
    world = bpy.data.worlds.new("world")
    bpy.context.scene.world = world
    world.use_nodes = True
    nodes = world.node_tree.nodes
    links = world.node_tree.links
    background = nodes.get("Background")
    if sky["kind"] == "nishita":
        texture = nodes.new("ShaderNodeTexSky")
        texture.sky_type = "NISHITA"
        texture.sun_elevation = math.radians(sky["sun_elevation_deg"])
        texture.sun_rotation = math.radians(sky["sun_rotation_deg"])
        texture.air_density = sky.get("air", 1.0)
        texture.dust_density = sky.get("dust", 1.0)
        links.new(texture.outputs["Color"], background.inputs["Color"])
        background.inputs["Strength"].default_value = sky.get("strength", 0.25)
        bpy.ops.object.light_add(type="SUN", location=(0.0, 0.0, 50.0))
        sun = bpy.context.active_object
        sun.data.energy = sky.get("sun_energy", 3.5)
        sun.data.angle = math.radians(0.6)
        elevation = math.radians(sky["sun_elevation_deg"])
        rotation = math.radians(sky["sun_rotation_deg"])
        # The sun lamp shines along its local -Z; point it from the sky's sun direction.
        sun.rotation_euler = Euler((math.pi / 2 - elevation, 0.0, rotation + math.pi / 2), "XYZ")
    elif sky["kind"] == "flat":
        background.inputs["Color"].default_value = srgb(sky["color"])
        background.inputs["Strength"].default_value = sky.get("strength", 1.0)
    else:
        raise ValueError(f"no sky named {sky['kind']!r}")


def add_lights(lights):
    for light in lights:
        bpy.ops.object.light_add(type="AREA", location=Vector(light["location"]))
        lamp = bpy.context.active_object
        lamp.name = light["id"]
        lamp.data.energy = light["energy"]
        lamp.data.size = light["size"]
        lamp.rotation_euler = Euler([math.radians(v) for v in light["rotation_deg"]], "XYZ")


def configure_render(render):
    scene = bpy.context.scene
    scene.render.engine = "CYCLES"
    scene.cycles.device = "CPU"
    scene.cycles.samples = render["samples"]
    scene.cycles.use_denoising = True
    scene.cycles.seed = render["seed"]
    scene.cycles.max_bounces = 6
    scene.render.threads_mode = "FIXED"
    scene.render.threads = render["threads"]
    scene.render.resolution_percentage = 100
    scene.render.image_settings.file_format = "PNG"
    scene.render.image_settings.color_mode = "RGB"
    scene.render.image_settings.color_depth = "8"
    scene.view_settings.view_transform = render.get("view_transform", "AgX")
    # A phone's rendering is contrasty and saturated; AgX's base look is neither.
    scene.view_settings.look = render.get("look", "AgX - Punchy")
    scene.view_layers[0].use_pass_z = True
    scene.use_nodes = True
    scene.render.use_compositing = True
    tree = scene.node_tree
    for node in list(tree.nodes):
        tree.nodes.remove(node)
    layers = tree.nodes.new("CompositorNodeRLayers")
    composite = tree.nodes.new("CompositorNodeComposite")
    tree.links.new(layers.outputs["Image"], composite.inputs["Image"])
    # The Viewer node's image is a 256 x 256 placeholder in background mode (measured on the
    # probe scene), so depth is written as a float EXR and read back below instead.
    depth_out = tree.nodes.new("CompositorNodeOutputFile")
    depth_out.name = "depth-out"
    depth_out.base_path = str(OUT_DIR / "_depth")
    depth_out.format.file_format = "OPEN_EXR"
    depth_out.format.color_depth = "32"
    depth_out.format.color_mode = "BW"
    depth_out.format.exr_codec = "ZIP"
    depth_out.file_slots[0].path = "depth_"
    tree.links.new(layers.outputs["Depth"], depth_out.inputs[0])


def render_camera(camera_spec):
    scene = bpy.context.scene
    moved = {}
    hidden = []
    for object_id in camera_spec.get("hide", []):
        obj = bpy.data.objects[object_id]
        hidden.append((obj, obj.hide_render))
        obj.hide_render = True
    for object_id, location in camera_spec.get("move", {}).items():
        obj = bpy.data.objects[object_id]
        moved[obj] = obj.location.copy()
        obj.location = Vector(location)
    for object_id in camera_spec.get("show", []):
        obj = bpy.data.objects[object_id]
        hidden.append((obj, obj.hide_render))
        obj.hide_render = False
    camera = scene.camera
    camera.location = Vector(camera_spec["location"])
    camera.rotation_mode = "XYZ"
    camera.rotation_euler = Euler([math.radians(v) for v in camera_spec["rotation_deg"]], "XYZ")
    camera.data.sensor_fit = "VERTICAL"
    camera.data.angle_y = math.radians(camera_spec["fov_y_deg"])
    camera.data.clip_start = 0.05
    camera.data.clip_end = 5000.0
    scene.render.resolution_x = camera_spec["width"]
    scene.render.resolution_y = camera_spec["height"]
    started = time.monotonic()
    image_path = OUT_DIR / f"{camera_spec['id']}.png"
    scene.render.filepath = str(image_path)
    import numpy

    depth_dir = OUT_DIR / "_depth"
    for stale in depth_dir.glob("depth_*.exr"):
        stale.unlink()
    bpy.ops.render.render(write_still=True)
    written = sorted(depth_dir.glob("depth_*.exr"))
    if len(written) != 1:
        raise RuntimeError(f"expected one depth EXR for {camera_spec['id']}, found {written}")
    loaded = bpy.data.images.load(str(written[0]))
    width, height = loaded.size
    if (width, height) != (camera_spec["width"], camera_spec["height"]):
        raise RuntimeError(f"depth is {width}x{height}, not the camera's own size")
    pixels = numpy.empty(width * height * loaded.channels, dtype=numpy.float32)
    loaded.pixels.foreach_get(pixels)
    # Bottom row first in Blender; keep channel 0 and flip to top row first.
    depth = numpy.ascontiguousarray(pixels.reshape(height, width, loaded.channels)[::-1, :, 0])
    bpy.data.images.remove(loaded)
    written[0].unlink()
    (OUT_DIR / f"{camera_spec['id']}.depth.f32").write_bytes(depth.astype("<f4").tobytes())
    elapsed = time.monotonic() - started
    for obj, previous in hidden:
        obj.hide_render = previous
    for obj, location in moved.items():
        obj.location = location
    world_from_camera = [list(row) for row in camera.matrix_world]
    return {
        "id": camera_spec["id"],
        "image": image_path.name,
        "depth": f"{camera_spec['id']}.depth.f32",
        "width": width,
        "height": height,
        "fov_y_deg": camera_spec["fov_y_deg"],
        "world_from_camera_blender": world_from_camera,
        "render_seconds": round(elapsed, 3),
    }


def main():
    OUT_DIR.mkdir(parents=True, exist_ok=True)
    bpy.ops.wm.read_factory_settings(use_empty=True)
    configure_render(SPEC["render"])
    build_world(SPEC["sky"])
    add_lights(SPEC.get("lights", []))
    for item in SPEC["objects"]:
        obj = add_shape(item)
        if item.get("hidden", False):
            obj.hide_render = True
    bpy.ops.object.camera_add(location=(0.0, 0.0, 1.6))
    bpy.context.scene.camera = bpy.context.active_object
    rendered = [render_camera(camera) for camera in SPEC["cameras"]]
    manifest = {
        "blender_version": bpy.app.version_string,
        "spec": SPEC_PATH.name,
        "cameras": rendered,
    }
    (OUT_DIR / "manifest.json").write_text(json.dumps(manifest, indent=1), encoding="utf-8")


main()
