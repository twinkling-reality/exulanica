"""Assemble the world-scale baseline record from the run files and the prose written after reading them.

    .venv/bin/python write_outcome.py [--replace]
        writes docs/evaluation/2026-09-22-world-scale-baseline.json

Every measured figure comes from outcome.py's sections, a run file, shading.json, point-counts.json
or the published manifest, and the prose interpolates those values. The few figures written into
departure and incident prose (the loads that motivated the gate, how often the browser hang was
reproduced) are observations made before the runs they explain and are not results. Integers only
(microseconds, millipixels, millionths, tenths of a percent, hundredths of a load, bytes), because
a digest input may hold no float. publish.py runs first, so the manifest this binds is the one on
disk; the record itself is not in it.
"""

def ms(us: int) -> str:
    return f"{us / 1000:,.1f} ms"


def sec(us: int) -> str:
    return f"{us / 1_000_000:,.1f} s"


def tenths(value: float | None) -> int | None:
    return None if value is None else round(value * 10)


def hundredths(values: list[float]) -> list[int]:
    return [round(v * 100) for v in values]


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def manifest() -> dict:
    return json.loads((HERE / "published-manifest.json").read_text())


def artifact_for(run: Path) -> str:
    return f"{ART}/runs/{run.parent.name}/{run.name}"

def frames(block: dict) -> dict:
    runs = block["per_run"]
    return {"pooled_p50_us": block["pooled_p50_us"], "pooled_p95_us": block["pooled_p95_us"],
            "pooled_max_us": block["pooled_max_us"], "pooled_frames": block["pooled_frames"],
            "pooled_frames_over_16700_us": block["pooled_over_16700"],
            "per_run_p50_us": [r["p50_us"] for r in runs],
            "per_run_p95_us": [r["p95_us"] for r in runs],
            "per_run_max_us": [r["max_us"] for r in runs],
            "per_run_interval_p95_us": [r["interval_p95_us"] for r in runs],
            "per_run_long_tasks": [r["long_tasks"] for r in runs],
            "draw_calls_p50": [r["draws_p50"] for r in runs],
            "runs_valid": block["valid"]}


def point(entry: dict, folder: str) -> dict:
    path = RUNS / folder / entry["file"]
    out = {"count": entry["count"],
           "run": {"artifact": artifact_for(path), "as_run_sha256": entry["file_sha256"]},
           "gate": {"idle_before_percent": entry["gate_pre_idle"],
                    "idle_during_mean_tenths": entry["gate_during_idle_mean_tenths"],
                    "idle_during_min_percent": entry["gate_during_idle_min"],
                    "idle_during_samples": entry["gate_during_samples"]},
           "load_one_minute_hundredths": hundredths([entry["load_start"], entry["load_end"]])}
    if "open" in entry:
        o = entry["open"]
        out["open"] = {"warm_first_frame_us": o["warm_first_frame_us"],
                       "warm_first_frame_median_us": o["warm_first_frame_median_us"],
                       "warm_complete_world_us": o["warm_complete_world_us"],
                       "warm_complete_world_median_us": o["warm_complete_world_median_us"],
                       "cold_first_frame_us": o["cold_first_frame_us"],
                       "cold_complete_world_us": o["cold_complete_world_us"],
                       "warm_document_to_form_median_us": o["warm_boot_to_form_median_us"],
                       "warm_submit_to_first_frame_median_us": o["warm_submit_to_first_frame_median_us"]}
    if "placeundo" in entry:
        p = entry["placeundo"]
        out["placement_and_undo"] = {
            "placement_ack_us": p["placement_ack_us"],
            "placement_ack_median_us": p["placement_ack_us_median"],
            "undo_ack_us": p["undo_ack_us"], "undo_ack_median_us": p["undo_ack_us_median"],
            "place_to_verdict_median_us": p["place_to_verdict_us_median"],
            "confirm_to_last_object_resident_median_us": p["last_object_resident_us_median"],
            "apply_server_median_us": p["apply_server_us_median"],
            "undo_server_median_us": p["undo_server_us_median"],
            "apply_response_bytes_median": p["apply_bytes_median"],
            "objects_after_each_undo": p["objects_after_undo"]}
    if "reload" in entry:
        r = entry["reload"]
        out["reload"] = {k: r[k] for k in ("identical", "version_same", "scene_same", "pixels_within",
                                           "scene_objects", "scene_maps", "control_share_millionths",
                                           "reload_share_millionths")}
    for key, name in (("walk", "walking"), ("look", "looking")):
        if key in entry:
            out[name] = frames(entry[key])
    return out

def estimates_series() -> dict:
    section = outcome.estimates()
    steps = section["steps"]
    counts = json.loads((RUNS / "estimates" / "point-counts.json").read_text())
    placements = [s for s in steps if s.get("step") == "place"]
    by_count = []
    for low, high in ((0, 1), (1, 4), (4, 16), (16, 32)):
        inside = [s for s in placements if low <= s["index"] < high]
        by_count.append({"estimates_before_from": low, "estimates_before_below": high,
                         "placements": len(inside),
                         "preview_p50_us": outcome.q([s["preview_us"] for s in inside], 0.5),
                         "apply_p50_us": outcome.q([s["apply_us"] for s in inside], 0.5),
                         "apply_max_us": max(s["apply_us"] for s in inside),
                         "apply_response_bytes_last": inside[-1]["apply_bytes"]})
    depth = []
    photographs = [s["pending"] for s in steps if s.get("step") == "uploaded"]
    for index, run in enumerate(outcome.depth_worker()):
        depth.append({"artifact": f"{ART}/runs/worker/{run['file']}", "as_run_sha256": run["file_sha256"],
                      "photographs": photographs[index],
                      "process_start_to_ready_ms": run["startup_seconds_milli"],
                      "ready_to_stopped_ms": run["stopped_seconds_milli"] - run["startup_seconds_milli"],
                      "peak_resident_kib": run["peak_rss_kib"], "exit": run["exit"],
                      "idle_during_mean_tenths": tenths(run["idle_during_mean"]),
                      "timing_valid": run["timing_valid"]})
    return {"made_through": PRE["series"]["estimates"]["made_through"],
            "points": [point(p, "estimates") for p in section["points"] if p.get("measured")],
            "placement_server": by_count,
            "point_maps": {k: counts[k] for k in ("point_maps", "point_count_min", "point_count_max",
                                                  "byte_size_min", "byte_size_max", "byte_size_total")},
            "point_map_container": "opm/2, 20 bytes a point (float32 position, four colour bytes, two uint16 tags)",
            "depth_runs": depth,
            "stopped": section["stopped"]}


def distance_series() -> dict:
    sessions = outcome.distance()["sessions"]
    shading = json.loads((RUNS / "distance" / "shading.json").read_text())
    marks: dict[int, dict] = {}
    gates = []
    for session in sessions:
        repeat = int(session["label"].split("-")[1])
        gates.append({"session": session["label"], "idle_before_percent": session["gate_pre_idle"],
                      "idle_during_mean_tenths": session["gate_during"]["mean_tenths"],
                      "idle_during_min_percent": session["gate_during"]["min"],
                      "artifact": f"{ART}/runs/distance/{session['file']}",
                      "as_run_sha256": session["file_sha256"]})
        legs = {leg["mark_m"]: leg for leg in session["legs"]}
        for ref in session["references"]:
            if ref["mark_m"] == 0:
                continue
            m = marks.setdefault(ref["mark_m"], {"mark_m": ref["mark_m"], "by_repeat": {}})
            leg = legs[ref["mark_m"]]
            face = next(s for s in shading if s["session"] == session["label"] and s["mark_m"] == ref["mark_m"])
            compared = ref["compared_with_mark_0"]
            m["by_repeat"][repeat] = {
                "registered_drawn_error_millipixels": ref["centroid_shift_millipixels"],
                "object_only_shift_millipixels": ref["object_only_centroid_shift_millipixels"],
                "iou_millionths": compared["iou_millionths"],
                "object_only_iou_millionths": compared["object_only_iou_millionths"],
                "object_free_frame_differing_millionths": compared["without_object_differing_millionths"],
                "sprint_seconds_milli": leg["seconds_milli"], "sprint_frames": leg["frames"],
                "sprint_work_p50_us": leg["work_p50_us"], "sprint_work_p95_us": leg["work_p95_us"],
                "sprint_work_max_us": leg["work_max_us"], "sprint_frames_over_16700_us": leg["over_16700"],
                "last_ten_seconds_work_p95_us": leg["last_ten_seconds_p95_us"],
                "sky_distinct_colours": face["sky_distinct_colours"],
                "sky_all_clear_colour": face["sky_all_clear_colour"],
                "lit_face_differing_pixels": face["face_differing_from_arrival"],
                "lit_face_middle_row_steps": face["face_middle_row_steps"]}
    out = []
    for mark in sorted(marks):
        rows = [marks[mark]["by_repeat"][r] for r in sorted(marks[mark]["by_repeat"])]
        entry = {"mark_m": mark, "added_after_registration": mark in (40, 80), "repeats": len(rows)}
        for key in rows[0]:
            entry[key] = [row[key] for row in rows]
        entry["registered_drawn_error_median_millipixels"] = int(statistics.median(
            entry["registered_drawn_error_millipixels"]))
        entry["object_only_shift_median_millipixels"] = int(statistics.median(
            entry["object_only_shift_millipixels"]))
        out.append(entry)
    arrival = [s for s in shading if s["mark_m"] == 0]
    return {"route": PRE["series"]["distance"]["route"], "marks": out,
            "arrival_sky_distinct_colours": sorted({s["sky_distinct_colours"] for s in arrival}),
            "lit_face_pixels": shading[0]["face_pixels"], "sessions": gates,
            "shading_artifact": f"{ART}/runs/distance/shading.json"}


def society_series() -> dict:
    out = {}
    for series in ("population", "history"):
        path = RUNS / "society" / f"{series}.json"
        raw = json.loads(path.read_text())
        points = []
        for p in raw["points"]:
            if "refused" in p:
                points.append({"population": p["population"], "refused_by_the_engine": p["refused"]})
                continue
            tick = p.get("tick") or p.get("ticks_since_last")
            points.append({
                "population": p["population"], "ticks_elapsed": p.get("ticks_elapsed", tick["n"]),
                "made_by_overriding_the_creation_constant": bool(p.get("patched", False)),
                "ticks_timed": tick["n"], "tick_p50_us": tick["p50_us"], "tick_p95_us": tick["p95_us"],
                "tick_max_us": tick["max_us"],
                "reload_p50_us": p["reload"]["p50_us"], "reload_p95_us": p["reload"]["p95_us"],
                "replay_us": p["replay_us"], "replay_median_us": p["replay"]["p50_us"],
                "replay_verified": p["replay_verified"],
                "reload_state_matches": p["reload_state_matches"],
                "state_bytes": p["facts"]["state_bytes"],
                "events_in_workspace": p["facts"]["events_in_workspace"]})
        out[series] = {"artifact": f"{ART}/runs/society/{series}.json", "as_run_sha256": sha(path),
                       "scope": "in-process server cost through SocietyRuntime and the society "
                                "repository; HTTP, serialisation and the permission floor excluded",
                       "profile": raw["profile"], "product_population": raw["product_population"],
                       "gate": {"idle_before_percent": raw["gate_pre"]["idle"],
                                "idle_during_mean_tenths": raw["gate_during"]["mean_tenths"],
                                "idle_during_min_percent": raw["gate_during"]["min"],
                                "idle_during_samples": raw["gate_during"]["samples"]},
                       "points": points}
        if series == "history":
            out[series]["all_ticks"] = raw["all_ticks"]
    return out

def knees(series: dict, estimates: dict, distance: dict, society: dict) -> dict:
    out = {}
    for layout in ("spread", "clustered"):
        k = outcome.object_knees(outcome.objects(layout))
        out[f"objects_{layout}"] = k
    k = outcome.object_knees(outcome.estimates())
    out["estimates"] = {m: v for m, v in k.items() if m not in ("placement_ack", "undo_ack")}
    registered = [m for m in distance["marks"] if not m["added_after_registration"]]
    out["distance"] = {
        "drawn_position_error": outcome.knee(
            [(m["mark_m"], m["registered_drawn_error_median_millipixels"],
              m["registered_drawn_error_millipixels"]) for m in registered],
            B["drawn_position_error_millipixels"]),
        "drawn_position_error_statistic": "median of the three repeats at a mark",
        "frame_work_while_sprinting": outcome.knee(
            [(m["mark_m"], max(m["sprint_work_p95_us"]), m["sprint_work_p95_us"]) for m in registered],
            B["frame_work_walking"])}
    population = [p for p in society["population"]["points"] if "tick_p95_us" in p]
    history = [p for p in society["history"]["points"] if "tick_p95_us" in p]
    out["society_population_at_60_ticks"] = {
        "tick_p95": outcome.knee([(p["population"], p["tick_p95_us"], []) for p in population],
                                 B["society_tick_p95"]),
        "reload_p95": outcome.knee([(p["population"], p["reload_p95_us"], []) for p in population],
                                   B["society_reload_p95"]),
        "replay": outcome.knee([(p["population"], p["replay_median_us"], p["replay_us"]) for p in population],
                               B["society_replay"]),
        "refused_counts": [p["population"] for p in society["population"]["points"]
                           if "refused_by_the_engine" in p]}
    replay = out["society_population_at_60_ticks"]["replay"]
    if replay.get("knee") and replay["between"][0] is None:
        replay["note"] = ("The smallest registered population, 32, was refused by the engine, so no "
                          "count below 128 was measured: the knee lies at or below 128, the one "
                          "population the product creates.")
    out["society_history_at_128"] = {
        "tick_p95": outcome.knee([(p["ticks_elapsed"], p["tick_p95_us"], []) for p in history],
                                 B["society_tick_p95"]),
        "reload_p95": outcome.knee([(p["ticks_elapsed"], p["reload_p95_us"], []) for p in history],
                                   B["society_reload_p95"]),
        "replay": outcome.knee([(p["ticks_elapsed"], p["replay_median_us"], p["replay_us"]) for p in history],
                               B["society_replay"])}
    return out


def crossing(low_count: int, low_us: int, high_count: int, high_us: int, budget: int) -> int:
    """Where the chord between two measured points meets the budget, in tenths of a count."""
    return round(10 * (low_count + (budget - low_us) * (high_count - low_count) / (high_us - low_us)))


def ranking(series: dict, estimates: dict, society: dict, knee: dict) -> dict:
    est = {p["count"]: p for p in estimates["points"]}
    e16 = est[16]["open"]["warm_first_frame_median_us"]
    e32 = est[32]["open"]["warm_first_frame_median_us"]
    e4 = est[4]["open"]["warm_first_frame_median_us"]
    est_cross = crossing(16, e16, 32, e32, B["open_first_frame"])
    per_estimate = (e32 - e16) // 16
    per_estimate_4_16 = (e16 - e4) // 12
    objects = {}
    for layout in ("spread", "clustered"):
        pts = {p["count"]: p["placement_and_undo"] for p in series[layout]["points"]}
        objects[layout] = {
            "placement_ack_crossing_tenths": crossing(200, pts[200]["placement_ack_median_us"], 1000,
                                                      pts[1000]["placement_ack_median_us"], B["placement_ack"]),
            "undo_ack_crossing_tenths": crossing(200, pts[200]["undo_ack_median_us"], 1000,
                                                 pts[1000]["undo_ack_median_us"], B["undo_ack"]),
            "ack_per_object_50_to_200_us": (pts[200]["placement_ack_median_us"] - pts[50]["placement_ack_median_us"]) // 150,
            "ack_per_object_200_to_1000_us": (pts[1000]["placement_ack_median_us"] - pts[200]["placement_ack_median_us"]) // 800}
    first_object = min(min(v["placement_ack_crossing_tenths"], v["undo_ack_crossing_tenths"]) for v in objects.values())
    history = {p["ticks_elapsed"]: p for p in society["history"]["points"] if "tick_p95_us" in p}
    replay_cross = crossing(1, history[1]["replay_median_us"], 60, history[60]["replay_median_us"], B["society_replay"])
    per_tick = (history[1440]["replay_median_us"] - history[60]["replay_median_us"]) // (1440 - 60)
    population = {p["population"]: p for p in society["population"]["points"] if "tick_p95_us" in p}
    sky_seconds_sprint_tenths = round(10 * SKY_RADIUS_M * 1000 / SPRINT_MM_S)
    sky_seconds_walk_tenths = round(10 * SKY_RADIUS_M * 1000 / WALK_MM_S)
    return {
        "rule": PRE["ranking"]["rule"],
        "assumed_rates": RATES,
        "first_limits": [
            {"rank": 1,
             "limit": "Opening a saved world that holds photo estimates takes longer than the 2 s budget.",
             "series": "estimates", "metrics": ["open_first_frame_us", "open_complete_world_us"],
             "budget_us": B["open_first_frame"],
             "knee": knee["estimates"]["open_first_frame"],
             "last_count_within": {"count": 16, "median_us": e16},
             "minutes_of_ordinary_use": {"from": 16 * RATES["estimate_seconds"] // 60,
                                         "to": 32 * RATES["estimate_seconds"] // 60,
                                         "at": f"{RATES['estimate_seconds']} s a placed estimate"},
             "chord_crossing_estimates_tenths": est_cross,
             "chord_crossing_minutes": est_cross * RATES["estimate_seconds"] // 600,
             "growth_per_estimate_us": {"from_4_to_16": per_estimate_4_16, "from_16_to_32": per_estimate},
             "register": "MEASURED knee; INFERRED crossing (straight line between the medians at 16 and 32)",
             "why_it_grows": (
                 "READ by the pre-registration in the code (loadAuthoredPointMaps runs before the "
                 "renderer mounts) and consistent with what was MEASURED: every placed estimate is "
                 "fetched and hashed before the renderer mounts, so the first "
                 "drawn frame waits for all of them and equals the complete world at every count "
                 f"(submit to first frame {ms(est[32]['open']['warm_submit_to_first_frame_median_us'])} "
                 f"of the {ms(e32)} at 32). The point maps are {estimates['point_maps']['byte_size_min']:,} "
                 f"to {estimates['point_maps']['byte_size_max']:,} bytes each."),
             "reproduction": (
                 "Place 32 photo estimates into one saved world on the endless ground through the "
                 "product (intake, the review granting the depth right, the depth worker, then Place "
                 "for each), reload the page and time the first drawn frame from the operator token's "
                 "submit; every one of five warm opens took over 2 s.")},
            {"rank": 2,
             "limit": "The objects panel's report that a placement or its take-back has landed arrives later than the 250 ms budget.",
             "series": "objects, both layouts", "metrics": ["placement_ack_us", "undo_ack_us"],
             "budget_us": B["placement_ack"],
             "knee": {"spread": knee["objects_spread"]["placement_ack"],
                      "clustered": knee["objects_clustered"]["placement_ack"],
                      "undo_spread": knee["objects_spread"]["undo_ack"],
                      "undo_clustered": knee["objects_clustered"]["undo_ack"]},
             "minutes_of_ordinary_use": {"from": 200 * RATES["object_placement_seconds"] // 60,
                                         "to": 1000 * RATES["object_placement_seconds"] // 60,
                                         "at": f"{RATES['object_placement_seconds']} s a placed object"},
             "chord_crossings": objects,

def distance_arrival_colours() -> list[int]:
    shading = json.loads((RUNS / "distance" / "shading.json").read_text())
    return [s["sky_distinct_colours"] for s in shading if s["mark_m"] == 0]


def distance_face(mark: int) -> list[int]:
    shading = json.loads((RUNS / "distance" / "shading.json").read_text())
    return [s["face_differing_from_arrival"] for s in shading if s["mark_m"] == mark]
