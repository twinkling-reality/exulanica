"""Run one timing-sensitive command inside quiet-slot (and gpu-slot with --gpu), with retries.

    python3 slot.py [--gpu] -- COMMAND...

Refuses when a product path differs from the measured commit. Inside the slots, gated.py refuses
unless CPU idle is at least 70 percent over 10 s, and the command discards itself if its mean idle
while running falls below 50 percent; either exits 75, and the run is retried after a pause outside
the slots, three attempts at most, as registered. Any other exit ends it.
"""

from __future__ import annotations

import subprocess
import sys
import time
from pathlib import Path

HERE = Path(__file__).resolve().parent
WORKTREE = HERE.parents[1]
BIN = Path("<MAIN_CHECKOUT>/.exulanica/bin")
COMMIT = "7e6144ea80847dd235a870ba14f4b7fbcf95dba0"
PRODUCT = ["exulanica", "web/packages", "migrations", "pyproject.toml", "web/package.json"]
PAUSE_SECONDS = 300


def product_unchanged() -> None:
    head = subprocess.run(["git", "rev-parse", "HEAD"], cwd=WORKTREE, capture_output=True,
                          text=True, check=True).stdout.strip()
    if head != COMMIT:
        raise SystemExit(f"HEAD is {head}, not the measured commit {COMMIT}")
    diff = subprocess.run(["git", "diff", "--quiet", "HEAD", "--", *PRODUCT], cwd=WORKTREE)
    status = subprocess.run(["git", "status", "--porcelain", "--untracked-files=all", "--", *PRODUCT],
                            cwd=WORKTREE, capture_output=True, text=True, check=True).stdout
    if diff.returncode != 0 or status.strip():
        raise SystemExit(f"a product path differs from {COMMIT}:\n{status}")


def main() -> None:
    arguments = sys.argv[1:]
    gpu = "--gpu" in arguments[: arguments.index("--")]
    command = arguments[arguments.index("--") + 1:]
    product_unchanged()
    # gated.py runs INSIDE the slots, so its idle readings are of the machine this run will have.
    wrapped = [str(BIN / "quiet-slot"), *([str(BIN / "gpu-slot")] if gpu else []),
               sys.executable, str(HERE / "gated.py"), "--", *command]
    for attempt in range(1, 4):
        started = time.strftime("%H:%M:%S")
        done = subprocess.run(wrapped, cwd=HERE)
        print(f"slot.py: attempt {attempt} started {started} exited {done.returncode}", flush=True)
        if done.returncode != 75:
            raise SystemExit(done.returncode)
        if attempt < 3:
            time.sleep(PAUSE_SECONDS)
    print("slot.py: three attempts refused for load; recorded as not measured", flush=True)
    raise SystemExit(75)


if __name__ == "__main__":
    main()
