// Repeatable 1280x720 browser evidence for the source-presentation correction.
// Usage: node capture_source_presentation.mjs <label> <url> <outputDirectory>
// Drives a fresh headless Chrome over the DevTools protocol: world arrival, source inspector, return.
import { spawn } from 'node:child_process';
import { mkdtempSync, writeFileSync, mkdirSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';

const [label, url, outDir] = process.argv.slice(2);
if (!label || !url || !outDir) throw new Error('usage: <label> <url> <outputDirectory>');
mkdirSync(outDir, { recursive: true });
const CHROME = '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome';
const profile = mkdtempSync(join(tmpdir(), 'orimera-capture-'));
const chrome = spawn(CHROME, [
  '--headless=new', '--remote-debugging-port=0', `--user-data-dir=${profile}`,
  '--window-size=1280,720', '--hide-scrollbars', '--no-first-run', '--no-default-browser-check',
  '--use-angle=swiftshader', '--enable-unsafe-swiftshader', '--force-device-scale-factor=1', 'about:blank',
], { stdio: ['ignore', 'ignore', 'pipe'] });
const wsUrl = await new Promise((resolve, reject) => {
  let buffer = '';
  chrome.stderr.on('data', (chunk) => {
    buffer += chunk.toString();
    const match = buffer.match(/DevTools listening on (ws:\/\/\S+)/);
    if (match) resolve(match[1]);
  });
  chrome.on('exit', (code) => reject(new Error(`chrome exited ${code}\n${buffer}`)));
  setTimeout(() => reject(new Error('chrome did not start')), 20000);
});
const ws = new WebSocket(wsUrl);
await new Promise((resolve, reject) => { ws.onopen = resolve; ws.onerror = reject; });
let nextId = 1;
const pending = new Map();
const consoleMessages = [];
let sessionId = null;
ws.onmessage = (event) => {
  const message = JSON.parse(event.data);
  if (message.id !== undefined) {
    const entry = pending.get(message.id);
    pending.delete(message.id);
    if (message.error) entry.reject(new Error(JSON.stringify(message.error))); else entry.resolve(message.result);
    return;
  }
  if (message.method === 'Runtime.consoleAPICalled') {
    consoleMessages.push({ type: message.params.type, text: message.params.args.map((a) => a.value ?? a.description ?? a.type).join(' ') });
  } else if (message.method === 'Runtime.exceptionThrown') {
    consoleMessages.push({ type: 'exception', text: message.params.exceptionDetails.text + ' ' + (message.params.exceptionDetails.exception?.description ?? '') });
  }
};
function send(method, params = {}, withSession = true) {
  const id = nextId++;
  const payload = { id, method, params };
  if (withSession && sessionId) payload.sessionId = sessionId;
  ws.send(JSON.stringify(payload));
  return new Promise((resolve, reject) => pending.set(id, { resolve, reject }));
}
const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
async function evaluate(expression) {
  const result = await send('Runtime.evaluate', { expression, returnByValue: true, awaitPromise: true });
  if (result.exceptionDetails) throw new Error(result.exceptionDetails.text + ' ' + JSON.stringify(result.exceptionDetails.exception));
  return result.result.value;
}
async function waitFor(expression, timeoutMs, description) {
  const started = Date.now();
  while (Date.now() - started < timeoutMs) {
    if (await evaluate(expression)) return Date.now() - started;
    await sleep(250);
  }
  throw new Error(`timeout waiting for ${description}`);
}
const captures = [];
async function screenshot(name) {
  // JPEG keeps the retained evidence small; the photographs themselves stay exact in the store.
  const { data } = await send('Page.captureScreenshot', { format: 'jpeg', quality: 88 });
  const file = join(outDir, `${label}-${name}.jpg`);
  writeFileSync(file, Buffer.from(data, 'base64'));
  const state = await evaluate(`(() => {
    const panel = document.querySelector('.reconstruction-availability');
    const inspector = document.querySelector('.reconstruction-inspector');
    const shell = document.getElementById('shell');
    const img = inspector?.querySelector('img');
    return {
      viewport: [innerWidth, innerHeight], devicePixelRatio,
      shellPrimary: shell?.getAttribute('data-primary') ?? null,
      panelVisible: panel !== null && getComputedStyle(panel).display !== 'none' && !panel.hidden,
      panelText: panel?.textContent ?? null,
      inspectorHidden: inspector?.hidden ?? null,
      inspectorTitle: inspector?.querySelector('h2')?.textContent ?? null,
      inspectorDisclosure: inspector?.querySelector('p')?.textContent ?? null,
      inspectorViewCount: inspector?.querySelector('select')?.options.length ?? null,
      inspectorState: inspector?.querySelector('[role=status]')?.textContent ?? null,
      inspectorImageLoaded: img ? img.complete && img.naturalWidth > 0 : null,
      inspectorImageSize: img ? [img.naturalWidth, img.naturalHeight] : null,
      statusSummaries: [...document.querySelectorAll('footer.status details summary')].map((s) => s.textContent),
      statusWarnings: [...document.querySelectorAll('footer.status .status-warning')].map((s) => s.textContent),
    };
  })()`);
  captures.push({ name, file, ...state });
  return state;
}
try {
  const { targetId } = await send('Target.createTarget', { url: 'about:blank' }, false);
  ({ sessionId } = await send('Target.attachToTarget', { targetId, flatten: true }, false));
  await send('Page.enable');
  await send('Runtime.enable');
  await send('Emulation.setDeviceMetricsOverride', { width: 1280, height: 720, deviceScaleFactor: 1, mobile: false });
  const started = Date.now();
  await send('Page.navigate', { url });
  const readyMs = await waitFor(`document.querySelector('canvas')?.dataset.companionRenderer === 'svg'`, 90000, 'Atlas mount');
  await sleep(4000); // let textures and the first frames settle
  const arrival = await screenshot('world-arrival');
  await evaluate(`document.querySelector('.reconstruction-availability button')?.click()`);
  await waitFor(`(() => { const i = document.querySelector('.reconstruction-inspector'); const img = i?.querySelector('img'); return i && !i.hidden && img && img.complete && img.naturalWidth > 0; })()`, 60000, 'inspector photograph');
  await sleep(500);
  const inspector = await screenshot('source-inspector');
  await evaluate(`[...document.querySelectorAll('.reconstruction-inspector button')].find((b) => b.textContent === 'Next view')?.click()`);
  await waitFor(`(() => { const i = document.querySelector('.reconstruction-inspector'); const img = i?.querySelector('img'); return i?.querySelector('select')?.selectedIndex === 1 && img && img.complete && img.naturalWidth > 0; })()`, 60000, 'second photograph');
  await sleep(300);
  const second = await screenshot('source-inspector-second');
  await evaluate(`[...document.querySelectorAll('.reconstruction-inspector button')].find((b) => b.textContent === 'Return to Atlas')?.click()`);
  await sleep(1200);
  const returned = await screenshot('world-after-return');
  const summary = {
    label, url, chrome: 'headless=new, swiftshader, 1280x720 css px, deviceScaleFactor 1',
    atlas_mount_ms: readyMs, total_ms: Date.now() - started, captures,
    console_errors: consoleMessages.filter((m) => m.type === 'error' || m.type === 'exception'),
    console_warnings: consoleMessages.filter((m) => m.type === 'warning'),
    checks: {
      arrival_panel_visible: arrival.panelVisible,
      arrival_inspector_hidden: arrival.inspectorHidden === true,
      inspector_opened_with_photograph: inspector.inspectorHidden === false && inspector.inspectorImageLoaded === true,
      inspector_titled_source_photographs: inspector.inspectorTitle === 'Source photographs',
      inspector_view_count: inspector.inspectorViewCount,
      second_photograph_loaded: second.inspectorImageLoaded === true,
      return_hides_inspector: returned.inspectorHidden === true,
      return_restores_panel: returned.panelVisible === true,
    },
  };
  writeFileSync(join(outDir, `${label}-capture.json`), JSON.stringify(summary, null, 2) + '\n');
  console.log(JSON.stringify(summary.checks, null, 2));
  console.log('console errors:', summary.console_errors.length, 'warnings:', summary.console_warnings.length);
} finally {
  ws.close();
  chrome.kill('SIGTERM');
  await sleep(500);
  rmSync(profile, { recursive: true, force: true });
}
