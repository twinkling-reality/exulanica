"""The standpoint join's pre-registered statistics (stage version 3), from one standpoint_eval.py run,
judged in pairs against version 1's run on the same sets in the same arm.

    python standpoint_gates.py <run.json> [<version-1-run.json> <preregistration.json>]

With a version 1 run and a pre-registration record, every gate the record states is judged with
the thresholds it states, read from the record rather than restated here; a gate that compares
with version 1 reads version 1's statistic from its run on the same sets. Without them, the
statistics are printed only, which is how the development data informed the record. The limit a
change must reach to be required of the join (``change_limit_ppm``) is read from the record, or
given with ``--change-limit-ppm`` when there is none yet.

A pair's class is decided from ground truth alone, never from the method's outcome:

* ``no_overlap``: under NO_OVERLAP of photograph a's sampled pixels are seen by photograph b;
* ``moved``: the two cameras stood MOVED_M or more apart;
* ``changed``: the changed parts' own surface covers at least the change limit of the pair's
  shared view, in either photograph (standpoint_eval.changed_surface_share);
* ``change_under_limit``: a changed part's own surface is in the shared view, under the limit.
  The join is not required to see it: such a pair may be joined or refused as changed, is never
  counted as joinable, and is reported by itself;
* ``joinable``: every other pair.

Versions 1 and 2 decided ``changed`` from a bounding sphere around each changed part plus 5 cm
at 2 percent (standpoint_eval.change_share), which counts the ground and walls around a part.
"""

from __future__ import annotations

import argparse
import json
import sys
from collections import Counter
from typing import Any

import numpy as np

NO_OVERLAP = 0.05
MOVED_M = 1.0
#: Pairs this much of whose view is shared are the ones a naming or joining rate is asked of. For
#: a changed pair, what counts is the share that did not change.
SHARED_ENOUGH = 0.20
#: The outcomes each class may end in without naming it wrongly.
ALLOWED = {
    "moved": {"moved_between_photographs", "insufficient_overlap"},
    "changed": {"scene_changed", "insufficient_overlap"},
    "change_under_limit": {"joined", "scene_changed", "insufficient_overlap", "inconsistent_with_set"},
    "joinable": {"joined", "insufficient_overlap", "inconsistent_with_set"},
    "no_overlap": {"insufficient_overlap", "joined", "inconsistent_with_set"},
}


def classify(pair: dict[str, Any], limit: float) -> str:
    truth = pair["truth"]
    if truth["overlap"] < NO_OVERLAP:
        return "no_overlap"
    if truth["distance_m"] >= MOVED_M:
        return "moved"
    share = truth["changed_surface_share"]
    if share >= limit:
        return "changed"
    if share > 0:
        return "change_under_limit"
    return "joinable"


def _percentiles(values: list[float]) -> dict[str, float]:
    array = np.array(values) if values else np.zeros(1)
    return {
        "n": len(values),
        "p50": round(float(np.percentile(array, 50)), 4),
        "p95": round(float(np.percentile(array, 95)), 4),
        "max": round(float(array.max()), 4),
    }


def statistics(run: dict[str, Any], limit: float) -> dict[str, Any]:
    arranged_excess: list[float] = []
    arranged_rows: list[dict[str, Any]] = []
    seam_p50: list[float] = []
    outcomes: dict[str, Counter] = {}
    misnamed: list[dict[str, Any]] = []
    under_limit: list[dict[str, Any]] = []
    moved_arranged = changed_arranged = 0
    moved_named = moved_asked = changed_named = changed_asked = 0
    joinable_asked = joinable_arranged = 0
    joinable_total = joinable_as_moved = joinable_as_changed = 0
    up_errors: list[float] = []
    up_methods: Counter = Counter()
    for set_result in run["sets"]:
        if "up_error_deg" in set_result:
            up_errors.append(set_result["up_error_deg"])
            up_methods[set_result["up"]["method"]] += 1
        joined = {m["ordinal"] for m in set_result["members"] if m["outcome"] == "joined"}
        for pair in set_result["pairs"]:
            klass = classify(pair, limit)
            outcomes.setdefault(klass, Counter())[pair["outcome"]] += 1
            if pair["outcome"] not in ALLOWED[klass]:
                misnamed.append({"set": set_result["set"], "a": pair["a"], "b": pair["b"], "class": klass, "outcome": pair["outcome"]})
            together = pair["a"] in joined and pair["b"] in joined
            shared = pair["truth"]["overlap"] >= SHARED_ENOUGH
            if together and pair.get("misregistration_deg"):
                parallax = pair["truth"]["irreducible_parallax_p95_deg"] or 0.0
                excess = pair["misregistration_deg"]["p95"] - parallax
                arranged_excess.append(excess)
                arranged_rows.append({"set": set_result["set"], "a": pair["a"], "b": pair["b"], "class": klass, "excess_deg": round(excess, 4)})
            if together and pair.get("depth_disagreement_seam"):
                seam_p50.append(pair["depth_disagreement_seam"]["p50"])
            if klass == "moved":
                moved_arranged += together
                if shared:
                    moved_asked += 1
                    moved_named += pair["outcome"] == "moved_between_photographs"
            elif klass == "changed":
                changed_arranged += together
                unchanged = pair["truth"]["overlap"] * (1 - pair["truth"]["changed_surface_share"])
                if unchanged >= SHARED_ENOUGH:
                    changed_asked += 1
                    changed_named += pair["outcome"] == "scene_changed"
            elif klass == "change_under_limit":
                under_limit.append({"set": set_result["set"], "a": pair["a"], "b": pair["b"], "outcome": pair["outcome"],
                                    "in_one_arrangement": together, "changed_surface_share": pair["truth"]["changed_surface_share"],
                                    "changed_ppm": pair["changed_ppm"]})
            elif klass == "joinable":
                joinable_total += 1
                joinable_as_moved += pair["outcome"] == "moved_between_photographs"
                joinable_as_changed += pair["outcome"] == "scene_changed"
                if shared:
                    joinable_asked += 1
                    joinable_arranged += together
    excess = np.array(arranged_excess) if arranged_excess else np.zeros(1)
    seams = np.array(seam_p50) if seam_p50 else np.zeros(1)
    return {
        "change_limit": limit,
        "outcomes_by_class": {k: dict(v) for k, v in sorted(outcomes.items())},
        "misnamed": misnamed,
        "change_under_limit": under_limit,
        "moved_named_changed": sum(1 for m in misnamed if m["class"] == "moved" and m["outcome"] == "scene_changed"),
        "changed_named_moved": sum(1 for m in misnamed if m["class"] == "changed" and m["outcome"] == "moved_between_photographs"),
        "pairs_in_one_arrangement": len(arranged_excess),
        "excess_misregistration_deg": {
            "p50": round(float(np.percentile(excess, 50)), 4),
            "p95": round(float(np.percentile(excess, 95)), 4),
            "max": round(float(excess.max()), 4),
        },
        "arranged": arranged_rows,
        "worst_pairs": sorted(arranged_rows, key=lambda row: -row["excess_deg"])[:5],
        "seam_depth_p50_by_pair": {
            "n": len(seam_p50),
            "median": round(float(np.median(seams)), 4),
            "p90": round(float(np.percentile(seams, 90)), 4),
            "max": round(float(seams.max()), 4),
        },
        "up_error_deg": _percentiles(up_errors),
        "up_methods": dict(up_methods),
        "moved_in_one_arrangement": int(moved_arranged),
        "moved_named": [moved_named, moved_asked],
        "changed_in_one_arrangement": int(changed_arranged),
        "changed_named": [changed_named, changed_asked],
        "joinable_refused_as_moved": [joinable_as_moved, joinable_total],
        "joinable_refused_as_changed": [joinable_as_changed, joinable_total],
        "joinable_in_one_arrangement": [joinable_arranged, joinable_asked],
    }


def paired(stats: dict[str, Any], v1: dict[str, Any]) -> dict[str, Any]:
    """What the two versions' runs on the same sets say side by side, and on the pairs both arranged."""
    mine = {(r["set"], r["a"], r["b"]): r["excess_deg"] for r in stats["arranged"]}
    theirs = {(r["set"], r["a"], r["b"]): r["excess_deg"] for r in v1["arranged"]}
    common = sorted(set(mine) & set(theirs))
    both = lambda values: round(float(np.percentile(np.array(values), 95)), 4) if values else 0.0  # noqa: E731
    return {
        "excess_misregistration_p95_deg": {"version_3": stats["excess_misregistration_deg"]["p95"],
                                           "version_1": v1["excess_misregistration_deg"]["p95"]},
        "pairs_in_one_arrangement": {"version_3": len(mine), "version_1": len(theirs), "both": len(common)},
        "excess_misregistration_p95_on_pairs_both_arranged_deg": {
            "version_3": both([mine[key] for key in common]), "version_1": both([theirs[key] for key in common])},
        "joins": {"version_3": stats["joinable_in_one_arrangement"], "version_1": v1["joinable_in_one_arrangement"]},
    }


def judge(stats: dict[str, Any], v1: dict[str, Any], all_gates: dict[str, Any], arm: str) -> dict[str, Any]:
    """Every gate the pre-registration states for this arm, with the threshold the record states."""
    gates = all_gates["arms"][arm]

    def rate(pair: list[int]) -> float:
        return 0.0 if pair[1] == 0 else pair[0] / pair[1]

    degrees = lambda value: value / 1000  # noqa: E731
    share = lambda value: value / 1_000_000  # noqa: E731
    lines_up = gates["lines_up"]
    verdicts = {
        "moved_never_joined": stats["moved_in_one_arrangement"] == 0,
        "changed_never_joined": stats["changed_in_one_arrangement"] == 0,
        "moved_never_named_changed": stats["moved_named_changed"] <= gates["names"]["moved_named_changed_max"],
        "changed_never_named_moved": stats["changed_named_moved"] <= gates["names"]["changed_named_moved_max"],
        "joinable_named_moved": stats["joinable_refused_as_moved"][0] <= gates["names"]["joinable_named_moved_max"],
        "joinable_named_changed": stats["joinable_refused_as_changed"][0] <= gates["names"]["joinable_named_changed_max"],
        "moved_named": rate(stats["moved_named"]) >= share(gates["named"]["moved_min_ppm"]),
        "changed_named": rate(stats["changed_named"]) >= share(gates["named"]["changed_min_ppm"]),
        "lines_up_p95_against_version_1": stats["excess_misregistration_deg"]["p95"]
        <= v1["excess_misregistration_deg"]["p95"] + degrees(lines_up["p95_over_version_1_max_millidegrees"]),
        "up_p50": stats["up_error_deg"]["p50"] <= degrees(gates["up"]["p50_max_millidegrees"]),
        "up_p95": stats["up_error_deg"]["p95"] <= degrees(gates["up"]["p95_max_millidegrees"]),
        "joins": rate(stats["joinable_in_one_arrangement"]) >= share(gates["joins"]["min_ppm"]),
        "joins_against_version_1": rate(stats["joinable_in_one_arrangement"])
        >= rate(v1["joinable_in_one_arrangement"]) - share(gates["joins"]["under_version_1_max_ppm"]),
        "seam_depth_median": stats["seam_depth_p50_by_pair"]["median"] <= share(gates["seam_depth"]["median_max_ppm"]),
        "seam_depth_p90": stats["seam_depth_p50_by_pair"]["p90"] <= share(gates["seam_depth"]["p90_max_ppm"]),
    }
    never = ("moved_never_joined", "changed_never_joined")
    return {"verdicts": verdicts, "never_passed": all(verdicts[name] for name in never),
            "passed": all(verdicts.values()), "paired_with_version_1": paired(stats, v1)}


def main(argv: list[str]) -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("run")
    parser.add_argument("version_1_run", nargs="?")
    parser.add_argument("preregistration", nargs="?")
    parser.add_argument("--change-limit-ppm", type=int)
    args = parser.parse_args(argv)
    run = json.loads(open(args.run, encoding="utf-8").read())
    prereg = json.loads(open(args.preregistration, encoding="utf-8").read())["record"] if args.preregistration else None
    if prereg is not None:
        limit_ppm = prereg["gates"]["change_limit_ppm"]
    elif args.change_limit_ppm is not None:
        limit_ppm = args.change_limit_ppm
    else:
        raise SystemExit("name the change limit: a pre-registration, or --change-limit-ppm")
    limit = limit_ppm / 1_000_000
    stats = statistics(run, limit)
    out: dict[str, Any] = {"arm": run["arm"], "policy_sha256": run["policy_sha256"], "stage_version": run.get("stage_version"), "statistics": stats}
    if args.version_1_run:
        v1_run = json.loads(open(args.version_1_run, encoding="utf-8").read())
        if v1_run["arm"] != run["arm"] or sorted(s["set"] for s in v1_run["sets"]) != sorted(s["set"] for s in run["sets"]):
            raise SystemExit("version 1's run is not of the same sets in the same arm")
        v1_stats = statistics(v1_run, limit)
        out["version_1"] = {"policy_sha256": v1_run["policy_sha256"], "stage_version": v1_run.get("stage_version"), "statistics": v1_stats}
        out["paired_with_version_1"] = paired(stats, v1_stats)
        if prereg is not None:
            out["judged"] = judge(stats, v1_stats, prereg["gates"], run["arm"])
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main(sys.argv[1:])
