"""Writes docs/artifacts/corridor/corridor-drawn.log.txt: what the tessellator draws of one baked
corridor tile, and what every record that does not draw is waiting for.

Run from the repository root, with a container and the tile document it was baked from:

    uv run python docs/artifacts/corridor/corridor-drawn.py.txt <tile.owd> <tile-document.json>

WHY IT IS SHAPED LIKE THIS. The container states one entry per record, and to say anything about
those entries you have to know which record each one is. That mapping is not the obvious one: the
container states its entries in ONE MERGED LIST SORTED BY KIND, while the document states owned
records and then halo records. Reading the order by eye gives a confident wrong answer. On
2026-09-17 it said a surface material record was drawing a carriageway, which is nonsense on its
face and is how the mistake was caught. So this reads both through the tessellator's own decoder
and document reader, by handing a small TypeScript program to the tessellator's own runtime, rather
than reimplementing either in Python. Python has no reader for the owd container and should not
grow a second one: two readers that disagree would be worse than none.

It reads no clock and no network, so two runs over one container agree.
"""

import json
import subprocess
import sys
import tempfile
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
TSX = ROOT / "web" / "node_modules" / ".bin" / "tsx"
OUTPUT = ROOT / "docs" / "artifacts" / "corridor" / "corridor-drawn.log.txt"

#: Run inside the tessellator's own runtime: decode the container, read the document, and answer
#: with the entries mapped to their records. It states no numbers of its own; the counting is below.
READER = """
import { readFileSync } from 'node:fs';
import { decodeOwd } from '<ROOT>/web/packages/loom-tess/src/core/owd.js';
import { readTileDocument } from '<ROOT>/web/packages/loom-tess/src/core/document.js';

const owd = decodeOwd(new Uint8Array(readFileSync(process.argv[2]))) as any;
const document = readTileDocument(new Uint8Array(readFileSync(process.argv[3]))) as any;
const records = document.grammars
  .flatMap((grammar: any) => [...grammar.owned, ...grammar.halo])
  .sort((a: any, b: any) => (a.kind < b.kind ? -1 : a.kind > b.kind ? 1 : a.version - b.version));

process.stdout.write(JSON.stringify({
  records: records.length,
  projections: Object.values<any>(owd.projections).map((projection: any) => ({
    name: projection.header.name,
    triangles: projection.header.triangle_count,
    vertices: projection.header.vertex_count,
    entries: projection.header.entries.map((entry: any) => ({
      kind: records[entry.record]?.kind ?? 'unknown record index',
      state: entry.state,
      needs: entry.needs ?? [],
      surfaces: (entry.surfaces ?? []).map((surface: any) => ({
        role: surface.role,
        orientation: surface.orientation,
        material: surface.material.state,
        triangles: surface.triangle_count,
      })),
    })),
  })),
}));
"""


def read(container: Path, document: Path) -> dict:
    if not TSX.exists():
        raise SystemExit("web/node_modules is not installed: pnpm install --frozen-lockfile --offline in web/")
    with tempfile.TemporaryDirectory() as work:
        program = Path(work) / "read-entries.ts"
        program.write_text(READER.replace("<ROOT>", str(ROOT)), encoding="utf-8")
        result = subprocess.run(
            [str(TSX), str(program), str(container), str(document)],
            capture_output=True, text=True, check=True,
        )
    return json.loads(result.stdout)


def counted(pairs) -> list[tuple[str, int]]:
    counts: dict[str, int] = {}
    for key in pairs:
        counts[key] = counts.get(key, 0) + 1
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))


def main() -> int:
    if len(sys.argv) != 3:
        print(__doc__)
        return 2
    container, document = Path(sys.argv[1]), Path(sys.argv[2])
    read_back = read(container, document)
    lines = [
        "what the tessellator draws of the corridor tile",
        "",
        f"container   {container.name}, {container.stat().st_size} bytes",
        f"document    {document.name}, {document.stat().st_size} bytes",
        f"records     {read_back['records']}",
        "",
    ]
    for projection in read_back["projections"]:
        entries = projection["entries"]
        lines.append(
            f"== {projection['name']}: {projection['triangles']} triangles, "
            f"{projection['vertices']} vertices, {len(entries)} entries"
        )
        for state, count in counted(entry["state"] for entry in entries):
            lines.append(f"   {count:>6}  {state}")
        drawn = [entry for entry in entries if entry["state"] == "drawn"]
        if drawn:
            lines.append("   drawn, by record kind:")
            for kind, count in counted(entry["kind"] for entry in drawn):
                lines.append(f"   {count:>6}  {kind}")
            surfaces = [
                f"{entry['kind']} {surface['role']} ({surface['material']}, {surface['orientation']})"
                for entry in drawn
                for surface in entry["surfaces"]
            ]
            dressed = sum(
                1 for entry in drawn for surface in entry["surfaces"] if surface["material"] == "record"
            )
            lines.append(
                f"   drawn surfaces: {len(surfaces)}, of which {dressed} dressed by a material "
                f"record and {len(surfaces) - dressed} with none"
            )
            for surface, count in counted(surfaces):
                lines.append(f"   {count:>6}  {surface}")
        waiting = [
            (need, entry["kind"])
            for entry in entries
            if entry["state"] == "unavailable"
            for need in (entry["needs"] or ["(states no need)"])
        ]
        if waiting:
            lines.append("   unavailable, by what each entry waits for:")
            for need, count in counted(need for need, _kind in waiting):
                lines.append(f"   {count:>6}  {need}")
                for kind, kind_count in counted(kind for this, kind in waiting if this == need):
                    lines.append(f"           {kind_count:>6}  {kind}")
        lines.append("")
    OUTPUT.write_text("\n".join(lines) + "\n", encoding="utf-8")
    print(OUTPUT, len(OUTPUT.read_bytes()), "bytes")
    return 0


if __name__ == "__main__":
    sys.exit(main())
