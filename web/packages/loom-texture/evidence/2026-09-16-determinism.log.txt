#!/bin/zsh
# Bake the texture catalog in five runs across three Node majors and two CPU architectures,
# then compare every file of every run against the first run with cmp, and against the
# committed assets/textures. Prints this script, then everything it observed.
set -u
WEB=/Users/glendonchin/dev/Technology/exulanica-preview-handoffs/web
OUT=$1
cd $WEB
echo "== script"
cat $0
echo
echo "== source"
echo "commit $(git -C .. rev-parse HEAD)"
echo "working tree changes under web/packages/loom-texture/src: $(git -C .. status --short -- web/packages/loom-texture/src | wc -l | tr -d ' ')"
npx tsc --build packages/loom-texture || exit 1
rm -rf $OUT && mkdir -p $OUT
run() {
  local name=$1; shift
  echo
  echo "== run $name"
  echo "runtime: $("$@" -p 'process.version + " " + process.arch + " " + process.platform')"
  "$@" ${RUNNER_ARGS[@]} --out $OUT/$name > $OUT/$name.stdout 2>&1
  echo "exit: $?"
  (cd $OUT/$name && find . -type f | LC_ALL=C sort | while read f; do shasum -a 256 "$f"; done)
}
RUNNER_ARGS=(node_modules/tsx/dist/cli.mjs packages/loom-texture/src/cli.ts)
run node24-arm64-a /opt/homebrew/opt/node@24/bin/node
run node24-arm64-b /opt/homebrew/opt/node@24/bin/node
run node26-arm64 /opt/homebrew/opt/node@26/bin/node
# esbuild ships a native binary per architecture, so the x86_64 run executes the tsc output.
RUNNER_ARGS=(packages/loom-texture/dist/src/cli.js)
run node20-arm64 arch -arm64 /usr/local/bin/node
run node20-x86_64-rosetta arch -x86_64 /usr/local/bin/node
echo
echo "== cmp, file by file, against node24-arm64-a"
FILES=($(cd $OUT/node24-arm64-a && find . -type f | LC_ALL=C sort))
echo "files per run: ${#FILES[@]}"
for other in node24-arm64-b node26-arm64 node20-arm64 node20-x86_64-rosetta; do
  differing=0
  for f in $FILES; do cmp -s "$OUT/node24-arm64-a/$f" "$OUT/$other/$f" || { echo "DIFFERS $other $f"; differing=$((differing+1)); }; done
  extra=$(cd $OUT/$other && find . -type f | wc -l | tr -d ' ')
  echo "$other: $differing of ${#FILES[@]} files differ; $extra files present"
done
echo
echo "== cmp against the committed assets/textures"
differing=0
for f in $FILES; do cmp -s "$OUT/node24-arm64-a/$f" "../assets/textures/$f" || { echo "DIFFERS committed $f"; differing=$((differing+1)); }; done
echo "committed: $differing of ${#FILES[@]} files differ; $(cd ../assets/textures && find . -type f | wc -l | tr -d ' ') files present"
