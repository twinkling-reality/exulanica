== script
#!/bin/zsh
# Bake one workspace request per published set on three runtimes across two CPU architectures,
# with the command the bake worker runs, and compare every container with cmp against the first
# run, every result line apart from the Node version, and the package source digest each run
# claimed against the one the backend computes. Prints this script and the request writer beside it,
# then everything it observed.
set -u
ROOT=/Users/glendonchin/dev/Technology/exulanica-preview-handoffs
WEB=$ROOT/web
HERE=${0:A:h}
WORK=$(mktemp -d $HERE/run.XXXXXX)
cd $WEB
echo "== script"
cat $0
echo
echo "== requests.mts, which the script runs from its own directory"
cat $HERE/requests.mts
echo
echo "== source"
echo "commit $(git -C $ROOT rev-parse HEAD)"
echo "working tree changes under web/packages/loom-texture and exulanica: $(git -C $ROOT status --short -- web/packages/loom-texture exulanica | wc -l | tr -d ' ')"
node_modules/.bin/tsc --build packages/loom-texture || exit 1
/opt/homebrew/opt/node@24/bin/node --import ./node_modules/tsx/dist/loader.mjs $HERE/requests.mts $WORK/requests || exit 1
REQUESTS=($(cd $WORK/requests && ls | LC_ALL=C sort))
echo "requests: ${#REQUESTS[@]}"
(cd $WORK/requests && for f in $REQUESTS; do echo "$(shasum -a 256 < $f | cut -d' ' -f1)  $f"; done)
run() {
  local name=$1; shift
  local out=$WORK/$name
  mkdir -p $out
  echo
  echo "== run $name"
  echo "runtime: $("$@" -p 'process.version + " " + process.arch + " " + process.platform')"
  for f in $REQUESTS; do
    "$@" ${RUNNER[@]} --request $WORK/requests/$f --out $out/${f%.json}.ltex > $out/${f%.json}.result 2> $out/${f%.json}.stderr
    echo "exit $? ${f%.json} $(shasum -a 256 < $out/${f%.json}.ltex | cut -d' ' -f1) $(wc -c < $out/${f%.json}.ltex | tr -d ' ')"
  done
}
RUNNER=(--import ./node_modules/tsx/dist/loader.mjs packages/loom-texture/src/workspace/cli.ts)
run node24-arm64 /opt/homebrew/opt/node@24/bin/node
run node26-arm64 /opt/homebrew/opt/node@26/bin/node
# esbuild ships a native binary per architecture, so the x86_64 run executes the tsc output.
RUNNER=(packages/loom-texture/dist/src/workspace/cli.js)
run node20-x86_64-rosetta arch -x86_64 /usr/local/bin/node
echo
echo "== cmp, container by container, against node24-arm64"
for other in node26-arm64 node20-x86_64-rosetta; do
  differing=0
  for f in $REQUESTS; do
    cmp -s $WORK/node24-arm64/${f%.json}.ltex $WORK/$other/${f%.json}.ltex || { echo "DIFFERS $other $f"; differing=$((differing+1)); }
  done
  echo "$other: $differing of ${#REQUESTS[@]} containers differ"
done
echo
echo "== result lines, with the Node version set aside"
python3 - $WORK "${REQUESTS[@]}" <<'PY'
import json, sys
from pathlib import Path
work, requests = Path(sys.argv[1]), sys.argv[2:]
runs = ["node24-arm64", "node26-arm64", "node20-x86_64-rosetta"]
digests = set()
for request in requests:
    stem = request[:-5]
    lines = []
    for run in runs:
        document = json.loads((work / run / f"{stem}.result").read_bytes())
        digests.add(document["runtime"]["package_sha256"])
        document["runtime"] = {"package_sha256": document["runtime"]["package_sha256"]}
        lines.append(json.dumps(document, sort_keys=True))
    print(f"{stem}: {'identical' if len(set(lines)) == 1 else 'DIFFERENT'} across {len(runs)} runs")
print(f"package source digests claimed: {sorted(digests)}")
PY
echo "package source digest the backend computes: $($ROOT/.venv/bin/python -c 'from exulanica.world.material_bakes import package_source_sha256; print(package_source_sha256())')"
rm -rf $WORK

== requests.mts, which the script runs from its own directory
// One bake request per published set, each the set's own recipe at its own resolution, under a
// fixed made-up recipe row id per set, written canonically to the directory given.
import { mkdirSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import {
  BAKE_REQUEST_PROFILE,
  LIBRARY,
  WORKSPACE_LICENCE_ID,
  canonicalBytes,
  sha256Hex,
  workspaceLicence,
} from '/Users/glendonchin/dev/Technology/exulanica-preview-handoffs/web/packages/loom-texture/src/index.ts';

const out = process.argv[2]!;
mkdirSync(out, { recursive: true });
const licence = workspaceLicence();
LIBRARY.forEach((source, index) => {
  const recipe = source.entry.recipe;
  const request = {
    profile: BAKE_REQUEST_PROFILE,
    set_id: `ws.${sha256Hex(new TextEncoder().encode(source.entry.set_id)).slice(0, 32)}`,
    version: 1,
    title: `Workspace variant of ${recipe.maker.id} version ${recipe.maker.version}`,
    summary: `Baked in one workspace from recipe ${sha256Hex(canonicalBytes(recipe))}.`,
    licence: { id: WORKSPACE_LICENCE_ID, sha256: licence.sha256 },
    maker_sha256: sha256Hex(canonicalBytes(source.maker.manifest)),
    recipe,
  };
  writeFileSync(join(out, `${String(index).padStart(2, '0')}-${source.entry.set_id}.json`), canonicalBytes(request));
});
