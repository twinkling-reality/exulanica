== script
#!/bin/zsh
# Export the committed dataset plan on three runtimes across two CPU architectures, compare every
# file of every run with cmp against the first, and the first run's manifest against the committed
# one. Prints this script, then everything it observed. The exported bytes are removed afterwards;
# the plan and the committed manifest rebuild them.
set -u
WEB=/Users/glendonchin/dev/Technology/exulanica-texture-dataset/web
DATA=$WEB/../.exulanica/datasets/texture
PLAN=packages/loom-texture/dataset/plans/texture-inverse-v1.json
COMMITTED=packages/loom-texture/dataset/manifests/texture-inverse-v1.json
cd $WEB
echo "== script"
cat $0
echo
echo "== source"
echo "commit $(git -C .. rev-parse HEAD)"
echo "working tree changes under web/packages/loom-texture and assets/textures: $(git -C .. status --short -- web/packages/loom-texture assets/textures | wc -l | tr -d ' ')"
npx tsc --build packages/loom-texture || exit 1
run() {
  local name=$1; shift
  local out=$DATA/evidence-$name
  rm -rf $out $out.stdout
  echo
  echo "== run $name"
  echo "runtime: $("$@" -p 'process.version + " " + process.arch + " " + process.platform')"
  "$@" ${RUNNER[@]} --plan $PLAN --out $out > $out.stdout 2>&1
  echo "exit: $?"
  (cd $out && find . -type f | LC_ALL=C sort | while read f; do shasum -a 256 "$f"; done)
}
RUNNER=(node_modules/tsx/dist/cli.mjs packages/loom-texture/src/dataset/cli.ts)
run node24-arm64 /opt/homebrew/opt/node@24/bin/node
run node26-arm64 /opt/homebrew/opt/node@26/bin/node
# esbuild ships a native binary per architecture, so the x86_64 run executes the tsc output.
RUNNER=(packages/loom-texture/dist/src/dataset/cli.js)
run node20-x86_64-rosetta arch -x86_64 /usr/local/bin/node
echo
echo "== cmp, file by file, against node24-arm64"
FILES=($(cd $DATA/evidence-node24-arm64 && find . -type f | LC_ALL=C sort))
echo "files per run: ${#FILES[@]}"
for other in node26-arm64 node20-x86_64-rosetta; do
  differing=0
  for f in $FILES; do cmp -s "$DATA/evidence-node24-arm64/$f" "$DATA/evidence-$other/$f" || { echo "DIFFERS $other $f"; differing=$((differing+1)); }; done
  echo "$other: $differing of ${#FILES[@]} files differ; $(cd $DATA/evidence-$other && find . -type f | wc -l | tr -d ' ') files present"
done
echo
echo "== the committed manifest"
if cmp -s "$DATA/evidence-node24-arm64/dataset.json" "$COMMITTED"; then echo "committed manifest: identical to node24-arm64's"; else echo "committed manifest: DIFFERS"; fi
echo "committed manifest sha256 $(shasum -a 256 < $COMMITTED | cut -d' ' -f1)"
rm -rf $DATA/evidence-node24-arm64 $DATA/evidence-node26-arm64 $DATA/evidence-node20-x86_64-rosetta
rm -f $DATA/evidence-*.stdout
