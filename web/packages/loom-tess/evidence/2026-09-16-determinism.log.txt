== script
#!/bin/zsh
# Bake the conformance tile through the Node entry and the browser entry, in five runs across
# three Node majors and two CPU architectures, and compare every container byte for byte and
# every digest against the checked-in golden literals. Then the two deliberate failures the brief
# asks for: a node:crypto import added to an existing core file, and one integer of the fixture
# moved. Both are restored from the committed tree. Prints this script, then what it observed.
set -u
REPO=/Users/glendonchin/dev/Technology/exulanica-personal-admission
WEB=$REPO/web
OUT=$1
cd $WEB
echo "== script"
cat $0
echo
echo "== source"
echo "commit $(git -C $REPO rev-parse HEAD)"
echo "working tree changes under web/packages/loom-tess: $(git -C $REPO status --short -- web/packages/loom-tess | wc -l | tr -d ' ')"
node_modules/.bin/tsc --build packages/loom-tess/tsconfig.json || exit 1
rm -rf $OUT && mkdir -p $OUT
FIXTURE=packages/loom-tess/test/fixtures/tile-conformance.json
cat > $OUT/preview.mjs <<'EOF'
// The browser entry, run under Node's Web Crypto, printing what the preview computed.
import { readFileSync } from 'node:fs';
const { previewTile } = await import(process.argv[2]);
const preview = await previewTile(new Uint8Array(readFileSync(process.argv[3])));
process.stdout.write(JSON.stringify({
  container_sha256: preview.containerSha256,
  tile_inputs_digest: preview.tileInputsDigest,
  triangle_digests: Object.fromEntries([...preview.triangleDigests].sort()),
}) + '\n');
EOF
run() {
  local name=$1; shift
  echo
  echo "== run $name"
  echo "runtime: $("$@" -p 'process.version + " " + process.arch + " " + process.platform')"
  "$@" packages/loom-tess/dist/src/node/cli.js bake $FIXTURE $OUT/$name.owd > $OUT/$name.node.json
  echo "node entry exit: $?"
  cat $OUT/$name.node.json
  "$@" $OUT/preview.mjs $WEB/packages/loom-tess/dist/src/browser/index.js $FIXTURE > $OUT/$name.browser.json
  echo "browser entry exit: $?"
  cat $OUT/$name.browser.json
  shasum -a 256 $OUT/$name.owd | sed "s#$OUT/##"
}
run node24-arm64-a /opt/homebrew/opt/node@24/bin/node
run node24-arm64-b /opt/homebrew/opt/node@24/bin/node
run node26-arm64 /opt/homebrew/opt/node@26/bin/node
run node20-arm64 arch -arm64 /usr/local/bin/node
run node20-x86_64-rosetta arch -x86_64 /usr/local/bin/node

echo
echo "== cmp, container by container, against node24-arm64-a"
for other in node24-arm64-b node26-arm64 node20-arm64 node20-x86_64-rosetta; do
  cmp $OUT/node24-arm64-a.owd $OUT/$other.owd && echo "$other: identical"
done
echo
echo "== digests against the golden literals in test/triangle-digest-conformance.test.ts"
GOLDEN_RENDER=$(grep -o "render_batch: '[0-9a-f]*'" packages/loom-tess/test/triangle-digest-conformance.test.ts | grep -o "[0-9a-f]\{64\}")
GOLDEN_NAV=$(grep -o "nav_envelope: '[0-9a-f]*'" packages/loom-tess/test/triangle-digest-conformance.test.ts | grep -o "[0-9a-f]\{64\}")
echo "golden render_batch $GOLDEN_RENDER"
echo "golden nav_envelope $GOLDEN_NAV"
for file in $OUT/*.json; do
  name=${file:t}
  render=$(grep -o '"render_batch":"[0-9a-f]*"' $file | grep -o "[0-9a-f]\{64\}")
  nav=$(grep -o '"nav_envelope":"[0-9a-f]*"' $file | grep -o "[0-9a-f]\{64\}")
  container=$(grep -o '"container_sha256":"[0-9a-f]*"' $file | grep -o "[0-9a-f]\{64\}")
  verdict=agrees
  [[ $render == $GOLDEN_RENDER && $nav == $GOLDEN_NAV ]] || verdict=DISAGREES
  echo "$name: $verdict, container $container"
done

echo
echo "== gate 1: node:crypto added to an existing core file"
cp packages/loom-tess/src/core/ascii.ts $OUT/ascii.ts.saved
{ echo "import { createHash } from 'node:crypto';"; echo "export const probe = createHash;"; cat $OUT/ascii.ts.saved; } > packages/loom-tess/src/core/ascii.ts
pnpm run typecheck 2>&1 | grep -E "error|ELIFECYCLE" | head -5
echo "typecheck exit ${pipestatus[1]}"
git -C $REPO checkout -- web/packages/loom-tess/src/core/ascii.ts
cmp packages/loom-tess/src/core/ascii.ts $OUT/ascii.ts.saved && echo "restored from the committed tree"
pnpm run typecheck > /dev/null 2>&1
echo "typecheck exit after restore $?"

echo
echo "== gate 3: one integer of the fixture moved"
node -e '
const fs = require("fs");
const path = process.argv[1];
const text = fs.readFileSync(path, "utf8");
const moved = text.replace("\"height_mm\":[-200,-80,40,160,-40,80,", "\"height_mm\":[-200,-80,40,160,-40,81,");
if (moved === text) throw new Error("the fixture no longer holds the height this probe moves");
fs.writeFileSync(path, moved);
' $FIXTURE
git -C $REPO diff --stat -- web/$FIXTURE
pnpm exec vitest run packages/loom-tess/test/triangle-digest-conformance.test.ts 2>&1 | grep -E "✓|×|Tests " | head -12
git -C $REPO checkout -- web/$FIXTURE
echo "restored: $(git -C $REPO status --short -- web/$FIXTURE | wc -l | tr -d ' ') changed files"
pnpm exec vitest run packages/loom-tess/test/triangle-digest-conformance.test.ts 2>&1 | grep -E "Tests " | head -2
echo
echo "== git status --short"
git -C $REPO status --short
