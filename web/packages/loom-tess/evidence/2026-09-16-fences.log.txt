== part 1 script
#!/bin/zsh
# Plant deliberate violations of the loom-tess fences, show each one fail, remove them, and show
# the fences green again. Every planted file is untracked and removed by name.
set -u
WEB=/Users/glendonchin/dev/Technology/exulanica-personal-admission/web
cd "$WEB"
echo "== head: $(git rev-parse HEAD)"
echo "== node: $(node --version)"

typeset -A PROBES
PROBES[packages/loom-tess/src/core/_probe_node_builtin.ts]="import { createHash } from 'node:crypto';
export const probe = createHash;"
PROBES[packages/loom-tess/src/core/_probe_workspace.ts]="import { canonicalJson } from '../../../scene-synth/src/format/index.js';
export const probe = canonicalJson;"
PROBES[packages/loom-tess/src/node/_probe_preview.ts]="import { previewTile } from '../browser/index.js';
export const probe = previewTile;"
PROBES[packages/loom-tess/src/browser/_probe_bake.ts]="import { nodeSha256 } from '../node/sha256.js';
export const probe = nodeSha256;"
PROBES[packages/app/src/_probe_tess_node.ts]="import { bakeDocumentFile } from '@exulanica/loom-tess/node';
export const probe = bakeDocumentFile;"
PROBES[packages/atlas-react/src/_probe_tess_relative.ts]="import { previewTile } from '../../loom-tess/src/browser/index.js';
export const probe = previewTile;"
PROBES[packages/atlas-react/src/_probe_tess_core_internal.ts]="import { decodeOwd } from '../../loom-tess/src/core/owd.js';
export const probe = decodeOwd;"
# Positive control: the core entry by its specifier. The only expected complaint is that
# atlas-react has not declared the dependency, which is the consuming lane's commit to make.
PROBES[packages/atlas-react/src/_probe_tess_core_entry.ts]="import { decodeOwd } from '@exulanica/loom-tess/core';
export const probe = decodeOwd;"

for file in ${(k)PROBES}; do
  if [[ -e $file ]]; then echo "refusing to overwrite $file"; exit 1; fi
done
for file in ${(k)PROBES}; do
  print -r -- "${PROBES[$file]}" > "$file"
  echo "== planted $file"
  cat "$file"
done

echo
echo "== pnpm run boundaries, with the probes"
pnpm run boundaries 2>&1 | grep -v '^$'
echo "== exit ${pipestatus[1]}"

echo
echo "== tsc -p packages/loom-tess/tsconfig.core.json --noEmit, with the probes"
node_modules/.bin/tsc -p packages/loom-tess/tsconfig.core.json --noEmit 2>&1
echo "== exit $?"
echo "== tsc -p packages/loom-tess/tsconfig.browser.json --noEmit, with the probes"
node_modules/.bin/tsc -p packages/loom-tess/tsconfig.browser.json --noEmit 2>&1
echo "== exit $?"

for file in ${(k)PROBES}; do rm -- "$file"; echo "== removed $file"; done

echo
echo "== pnpm run boundaries, probes removed"
pnpm run boundaries 2>&1 | grep -v '^$'
echo "== exit ${pipestatus[1]}"
echo "== tsc -p packages/loom-tess/tsconfig.core.json --noEmit, probes removed"
node_modules/.bin/tsc -p packages/loom-tess/tsconfig.core.json --noEmit 2>&1
echo "== exit $?"
echo "== git status --short"
git status --short
