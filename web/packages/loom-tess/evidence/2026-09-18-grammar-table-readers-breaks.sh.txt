#!/bin/zsh
# Break one claim at a time on a committed tree, run the WHOLE loom-tess package, and print which
# tests failed and that the count is not zero. Restores with reset --hard plus clean and PRINTS the
# restored state, because a restore command's success reports only that it had nothing to object to.
emulate -L zsh
set -u
ROOT=/Users/glendonchin/dev/Technology/exulanica-city-vocabulary
cd $ROOT || exit 1

dirty=$(git status --porcelain | wc -l | tr -d ' ')
if [[ $dirty != 0 ]]; then
  print -r -- "REFUSING: the tree is dirty ($dirty paths); a break needs a committed tree"
  exit 1
fi
print -r -- "baseline $(git rev-parse --short HEAD), clean"

run_break() {
  local label="$1" file="$2" from="$3" to="$4"
  print -r -- ""
  print -r -- "================ BREAK: $label"
  python3 - "$ROOT/$file" "$from" "$to" <<'PY'
import sys
path, frm, to = sys.argv[1], sys.argv[2], sys.argv[3]
text = open(path).read()
n = text.count(frm)
if n != 1:
    print(f"PLANT FAILED: the text to replace occurs {n} times in {path}")
    sys.exit(3)
open(path, "w").write(text.replace(frm, to))
print(f"planted in {path}")
PY
  if [[ $? != 0 ]]; then
    git reset -q --hard HEAD && git clean -qfd
    print -r -- "restored: $(git status --porcelain | wc -l | tr -d ' ') dirty paths"
    return 1
  fi
  print -r -- "tree while broken: $(git status --porcelain | tr '\n' ' ')"
  local log=/private/tmp/claude-501/-Users-glendonchin-dev-Technology-orimera/45065fe9-6c90-424b-88a8-83d38894da72/scratchpad/break-$label.log.txt
  (cd $ROOT/web && pnpm exec vitest run packages/loom-tess) > $log 2>&1
  local rc=$?
  print -r -- "vitest exit $rc"
  grep -E '^\s+(Test Files|Tests)\s' $log
  local failed
  failed=$(grep -c '^\s*×' $log)
  print -r -- "FAILED TEST COUNT: $failed"
  if [[ $failed == 0 ]]; then
    print -r -- "!!! THE BREAK ASKED NOTHING: no test failed. Suspect the harness before believing it."
  else
    grep '^\s*×' $log | sed 's/^/    /' | head -20
  fi
  git reset -q --hard HEAD && git clean -qfd
  print -r -- "restored: $(git status --porcelain | wc -l | tr -d ' ') dirty paths, HEAD $(git rev-parse --short HEAD)"
}

run_break positional-document web/packages/loom-tess/src/core/document.ts \
  "  const tileTable = tileTableOf(GRAMMAR_TABLES, document.tile, 'tile');" \
  "  const tileTable = GRAMMAR_TABLES[0]!;"

run_break positional-container web/packages/loom-tess/src/core/owd.ts \
  "    const table = tileTableOf(GRAMMAR_TABLES, header.tile, 'tile');" \
  "    const table = GRAMMAR_TABLES[0]!;"

run_break no-agreement-check web/packages/loom-tess/src/core/record-shapes.ts \
  "  for (const value of stated) {
    if (value.length !== agreed.length) throw new ShapeTableError(\`the grammars this tessellator reads state different \${what}\`);
    if (value.some((item, index) => item !== agreed[index])) throw new ShapeTableError(\`the grammars this tessellator reads state different \${what}\`);
  }" \
  "  // BREAK: the first table's answer, whatever the others state."

run_break record-versions-overwrite web/packages/loom-tess/src/core/bake.ts \
  "      const already = versions[shape.kind];
      if (already !== undefined && already !== shape.version) {
        throw new RangeError(\`\${shape.kind} is read at versions \${already} and \${shape.version}, and this states one version a kind\`);
      }
" \
  ""

run_break bound-disagreement web/packages/loom-tess/src/core/bake.ts \
  "  if (bounds.length !== 1) throw new RangeError(\`the grammars this tessellator reads fix \${name} at \${JSON.stringify(bounds)}, and a bake states one\`);" \
  "  // BREAK: the first bound, whatever the others fix."

run_break two-pins-take-first web/packages/loom-tess/src/core/document.ts \
  "  if (stated.length > 1) {
    fail(at, \`names \${stated.length} pins for \${JSON.stringify(TILE_GRAMMAR_ID)}, and the tile record is written in one version\`);
  }" \
  "  // BREAK: the first pin that names it."

run_break no-pin-unchecked web/packages/loom-tess/src/core/document.ts \
  "  if (stated.length === 0) {
    fail(at, \`names no pin for \${JSON.stringify(TILE_GRAMMAR_ID)}, whose version states the shape of the tile record itself\`);
  }" \
  "  // BREAK: no check that the record names its own grammar at all."

print -r -- ""
print -r -- "FINAL TREE: $(git rev-parse --short HEAD), $(git status --porcelain | wc -l | tr -d ' ') dirty paths"
